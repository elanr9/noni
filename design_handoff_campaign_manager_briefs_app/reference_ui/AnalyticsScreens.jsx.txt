function aRangeData(range, weekly) {
  const lastW = weekly[weekly.length - 1] || 0;
  const wave = (n, base, amp, rise) => Array.from({ length: n }, (_, i) => Math.max(0.1, +(base * (1 + amp * Math.sin(i * 1.35 + 0.8) + rise * i / n)).toFixed(1)));
  if (range === 'Last 24 hours') return { data: wave(12, lastW / 7 / 10, 0.45, 0.5), labels: ['2a', '10a', '6p'] };
  if (range === 'Last 7 days') return { data: wave(7, lastW / 7, 0.3, 0.25), labels: ['Wed', 'Sat', 'Tue'] };
  if (range === 'Last 2 weeks') return { data: wave(14, lastW / 7, 0.35, 0.3), labels: ['Jul 30', 'Aug 5', 'Aug 11'] };
  if (range === 'Last month') return { data: weekly.slice(-5), labels: ['Jul 13', 'Jul 27', 'Aug 10'] };
  return { data: weekly, labels: ['May 25', 'Jul 6', 'Aug 3'] };
}

function MAreaChart({ series, labels }) {
  const W = 350, H = 170, P = { t: 12, r: 8, b: 22, l: 34 };
  const max = Math.max(...series) * 1.15;
  const iw = W - P.l - P.r, ih = H - P.t - P.b;
  const pts = series.map((v, i) => [P.l + iw * i / (series.length - 1), P.t + ih * (1 - v / max)]);
  const line = pts.map((p, i) => (i ? 'L' : 'M') + p[0].toFixed(1) + ' ' + p[1].toFixed(1)).join(' ');
  const last = pts[pts.length - 1];
  return (
    <svg viewBox={`0 0 ${W} ${H}`} style={{ width: '100%', height: 'auto', display: 'block' }}>
      {[0, 0.5, 1].map((f, i) => (
        <g key={i}>
          <line x1={P.l} x2={W - P.r} y1={P.t + ih * f} y2={P.t + ih * f} stroke="var(--line)" strokeWidth="1"></line>
          <text x={P.l - 7} y={P.t + ih * f + 4} textAnchor="end" style={{ font: '600 10px var(--font-ui)', fill: 'var(--slate-400)' }}>{aK(max * (1 - f) * 1000)}</text>
        </g>
      ))}
      <path d={`${line} L ${(P.l + iw).toFixed(1)} ${P.t + ih} L ${P.l} ${P.t + ih} Z`} fill="rgba(27,166,238,0.10)"></path>
      <path d={line} fill="none" stroke="var(--blue-500)" strokeWidth="2.5" strokeLinejoin="round" strokeLinecap="round"></path>
      <circle cx={last[0]} cy={last[1]} r="4.5" fill="var(--blue-500)" stroke="#fff" strokeWidth="2"></circle>
      {labels.map((w, i) => <text key={w + i} x={P.l + iw * (labels.length > 1 ? i / (labels.length - 1) : 0)} y={H - 6} textAnchor={i === 0 ? 'start' : i === labels.length - 1 ? 'end' : 'middle'} style={{ font: '600 10px var(--font-ui)', fill: 'var(--slate-400)' }}>{w}</text>)}
    </svg>
  );
}

/* Anchored menu pill — same behaviour as the web console dropdowns. */
function MMenu({ trigger, active, children }) {
  const { Icon } = window.NoniDesignSystem_710e43;
  const [open, setOpen] = React.useState(false);
  const ref = React.useRef(null);
  React.useEffect(() => {
    const out = (e) => { if (ref.current && !ref.current.contains(e.target)) setOpen(false); };
    window.addEventListener('pointerdown', out);
    return () => window.removeEventListener('pointerdown', out);
  }, []);
  return (
    <div ref={ref} style={{ position: 'relative', display: 'inline-block' }}>
      <button type="button" onClick={() => setOpen(!open)}
        style={{ display: 'inline-flex', alignItems: 'center', gap: 6, padding: '9px 13px', borderRadius: 999, border: '1px solid var(--border)', cursor: 'pointer', whiteSpace: 'nowrap', boxShadow: 'var(--shadow-card)', background: active ? 'var(--blue-100)' : 'var(--white)', font: '700 13px var(--font-ui)', color: active ? 'var(--blue-700)' : 'var(--ink)' }}>
        {trigger}<Icon name="chevron-down" size={13} color={active ? 'var(--blue-700)' : 'var(--slate-400)'} />
      </button>
      {open ? (
        <div style={{ position: 'absolute', top: 'calc(100% + 6px)', right: 0, zIndex: 80, minWidth: 180, background: 'var(--white)', border: '1px solid var(--border)', borderRadius: 14, boxShadow: 'var(--shadow-raised)', padding: 6 }}>
          {children((fn) => { fn(); setOpen(false); })}
        </div>
      ) : null}
    </div>
  );
}

                          {['All creators', ...A_CREATORS.map((p) => p.name)].map((c) => <MItem key={c} label={c} on={creatorF === c} pick={() => close(() => setCreatorF(c))} />)}
                        </>
                      )}
                    </MMenu>
                    <MMenu trigger="Sort by">
                      {(close) => A_SORTS.map((s) => <MItem key={s} label={s} on={sortBy === s} pick={() => close(() => setSortBy(s))} />)}
                    </MMenu>
                    <MMenu trigger={range}>
                      {(close) => A_RANGES.map((r) => <MItem key={r} label={r} on={range === r} pick={() => close(() => setRange(r))} />)}
                    </MMenu>
                  </div>
                  {sortBy === 'Views over time' ? (
                    <>
                      <SectionLabel style={{ padding: '0 0 10px' }}>Views · {range.toLowerCase()}{cut}</SectionLabel>
                      <MAreaChart series={chart.data} labels={chart.labels} />
                    </>) : null}
                  {sortBy === 'Top creators' ? (
                    <>
                      <SectionLabel style={{ padding: 0 }}>Top creators{cut}</SectionLabel>
                      {ranked.map((p, i) => <RankRow key={p.id} rank={i + 1} name={p.name} views={p.v} max={maxCr} onOpen={() => onOpenCreator && onOpenCreator(p.id)} />)}
                    </>) : null}
                  {sortBy === 'Top posts' ? (
                    <>
                      <SectionLabel style={{ padding: 0 }}>Top posts{cut}</SectionLabel>
                      {posts.length === 0 ? <p style={{ margin: '10px 0 2px', font: '600 13px var(--font-ui)', color: 'var(--slate-400)' }}>No posts match these filters.</p>
                        : posts.map((q, i) => <APostRow key={q.id} q={q} border={i < posts.length - 1} onOpen={() => setPost(q)} />)}
                    </>) : null}
                  {sortBy === 'Formats' ? (
                    <>
                      <SectionLabel style={{ padding: '0 0 12px' }}>Posts by format{crObj ? ' · ' + crObj.name.split(' ')[0] : ''}</SectionLabel>
                      <div style={{ display: 'flex', flexDirection: 'column', gap: 13 }}>
                        {fmtEntries.map(([k, v]) => (
                          <div key={k} style={{ display: 'flex', alignItems: 'center', gap: 10 }}>
                            <span style={{ width: 74, font: '600 12.5px var(--font-ui)', color: 'var(--slate-500)' }}>{k}s</span>
                            <span style={{ flex: 1, height: 9, borderRadius: 999, background: 'var(--fill-quiet)', overflow: 'hidden' }}>
                              <span style={{ display: 'block', width: Math.round(100 * v / maxFmt) + '%', height: '100%', borderRadius: 999, background: 'var(--blue-500)' }}></span>
                            </span>
                            <span style={{ width: 34, textAlign: 'right', font: '700 12.5px var(--font-ui)', color: 'var(--ink)' }}>{v}</span>
                          </div>
                        ))}
                      </div>
                    </>) : null}
                </>
              )}
            </Card>
          ) : (
            <>
              <Card pad={12}>
                <SectionLabel style={{ padding: '0 0 8px' }}>Daily activity · August</SectionLabel>
                <MMonthCal selected={day} onPick={(d) => { setDay(d); setPost(null); }} />
              </Card>
              {day ? (
                <Card pad={12}>
                  {post ? <MPostDetail post={post} onBack={() => setPost(null)} /> : <MDayDetail day={day} onOpenPost={setPost} onClose={() => setDay(null)} />}
                </Card>) : null}
            </>
          )}
        </>
      ) : null}
    </AdminScreen>
  );
}

      <span style={{ display: 'block', width: 24, height: 24, borderRadius: 999, background: 'var(--white)', boxShadow: 'var(--shadow-card)', transform: on ? 'translateX(18px)' : 'translateX(0)', transition: 'transform var(--dur-fast) var(--ease-out)' }}></span>
    </button>
  );
}
