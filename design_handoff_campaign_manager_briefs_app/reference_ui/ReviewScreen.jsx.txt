/* Review — the daily loop. Three queues in one tab: post submissions,
   music approvals (slideshows only), and creator account approval.
   Counts live in the switcher so an empty lane never costs a tap. */

function ReviewScreen({ state = 'default', lane = 0, onOpen, onOpenMusic, onOpenAccount }) {
  const { Button, EmptyState, Icon } = window.NoniDesignSystem_710e43;
  const A = window.NONI_ADMIN;
  const [tab, setTab] = React.useState(lane);
  React.useEffect(() => setTab(lane), [lane]);

  const loading = state === 'loading';
  const empty = state === 'empty';
  const subs = empty ? [] : A.submissions;
  const music = empty ? [] : A.music;
  const accounts = empty ? [] : A.accounts.filter((a) => a.state === 'pending');
  const creator = (id) => A.creators.find((c) => c.id === id);

  const total = subs.length + music.length + accounts.length;

  return (
    <AdminScreen>
      <AdminHeader
        title="Review"
        subtitle={loading ? null : total === 0 ? 'Everything is cleared. Creators are recording the rest of the week.' : total === 1 ? "One to clear, then you're done for today." : "Approve posts and they will be posted automatically!"}
        right={loading ? <SkeletonLine w={84} h={30} /> : <CountPill tone={total === 0 ? 'clear' : 'brand'}>{total === 0 ? 'All clear' : `${total} waiting`}</CountPill>}
      />

      <Segmented
        active={tab} onSelect={setTab}
        items={[{ label: 'Posts', count: loading ? '' : subs.length }, { label: 'Music', count: loading ? '' : music.length }, { label: 'Accounts', count: loading ? '' : accounts.length }]}
        style={{ marginTop: 4 }}
      />

      {loading ? <SkeletonRows n={4} height={96} /> : null}

      {!loading && tab === 0 ? (
        subs.length ? (
          <div style={{ display: 'flex', flexDirection: 'column', gap: 10 }}>
            {subs.map((s) => <SubmissionRow key={s.id} item={s} creator={creator(s.creator)} onClick={() => onOpen && onOpen(s)} />)}
            <p style={{ margin: '6px 2px 0', font: '400 13px/1.45 var(--font-ui)', color: 'var(--slate-400)' }}>
              Reject a single clip and only that clip goes back. The rest stay approved.
            </p>
          </div>
        ) : (
          <EmptyState icon="inbox" title="Nothing to review" body="Creators are recording this week's posts. New submissions land here, newest first." style={{ marginTop: 40 }} />
        )
      ) : null}

      {!loading && tab === 1 ? (
        music.length ? (
          <div style={{ display: 'flex', flexDirection: 'column', gap: 10 }}>
            <p style={{ margin: '2px 2px 0', font: '400 13px/1.45 var(--font-ui)', color: 'var(--slate-500)' }}>
              Slideshows only. Open the post, check the song is on it, approve. Approval unlocks that post's earnings.
            </p>
            {music.map((m) => <MusicApprovalRow key={m.id} item={m} creator={creator(m.creator)} onOpen={() => onOpenMusic && onOpenMusic(m)} />)}
          </div>
        ) : (
          <EmptyState icon="music-2" title="No songs waiting" body="Creators tap Music added once the track is on a live slideshow. It lands here." style={{ marginTop: 40 }} />
        )
      ) : null}

      {!loading && tab === 2 ? (
        accounts.length ? (
          <div style={{ display: 'flex', flexDirection: 'column', gap: 10 }}>
            {accounts.map((a) => <AccountRow key={a.id} item={a} creator={creator(a.creator)} onClick={() => onOpenAccount && onOpenAccount(a)} />)}
            <SectionLabel>Sent back</SectionLabel>
            {A.accounts.filter((a) => a.state === 'needs_changes').map((a) => (
              <AccountRow key={a.id} item={a} creator={creator(a.creator)} onClick={() => onOpenAccount && onOpenAccount(a)} />
            ))}
          </div>
        ) : (
          <EmptyState icon="circle-user-round" title="No accounts to approve" body="Every creator on the roster is linked. New creators show up here after they upload their warm-up proof." style={{ marginTop: 40 }} />
        )
      ) : null}
    </AdminScreen>
  );
}

/* Every row is the same height: 72px thumb plus 12px padding, so the queue
   scrolls as an even rhythm. The retake count rides on the thumbnail rather
   than the chip row, which would push a card taller. */
function SubmissionRow({ item, creator, onClick }) {
  return (
    <Card pad={12} onClick={onClick} style={{ display: 'flex', gap: 12, alignItems: 'stretch', height: 96, boxSizing: 'border-box' }}>
      <Thumb format={item.format} w={54} h={72} badge={window.isReel(item.format) ? item.duration : `${item.clips}`} topBadge={item.attempt > 1 ? `Take ${item.attempt}` : null} />
      <div style={{ flex: 1, minWidth: 0, display: 'flex', flexDirection: 'column', justifyContent: 'space-between' }}>
        <div style={{ display: 'flex', alignItems: 'center', gap: 7 }}>
          <Avatar name={creator.name} size={20} photo />
          <span style={{ font: '700 13px var(--font-ui)', color: 'var(--ink)' }}>{creator.short}</span>
          <span style={{ flex: 1 }} />
          <span style={{ flex: '0 0 auto', whiteSpace: 'nowrap', font: '600 12px var(--font-ui)', color: 'var(--slate-400)' }}>{item.age}</span>
        </div>
        <div style={{ font: '700 15px/1.28 var(--font-display)', letterSpacing: '-0.2px', color: 'var(--ink)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{item.title}</div>
        <div style={{ display: 'flex', gap: 6, alignItems: 'center', overflow: 'hidden' }}>
          <FormatChip format={item.format} size="sm" />
          <TypeChip>{item.typeLabel}</TypeChip>
          <TypeChip>{window.isReel(item.format) ? `${item.clips} clips` : `${item.clips} slides`}</TypeChip>
        </div>
      </div>
    </Card>
  );
}

/* One tap after a glance. The song is the only question. */
function MusicApprovalRow({ item, creator, onOpen, approved = false }) {
  const { Button, Icon } = window.NoniDesignSystem_710e43;
  return (
    <Card pad={12} onClick={onOpen} style={{ display: 'flex', gap: 12, alignItems: 'center' }}>
      <Thumb format={item.format} w={44} h={58} />
      <div style={{ flex: 1, minWidth: 0 }}>
        <div style={{ font: '700 14px/1.3 var(--font-display)', letterSpacing: '-0.2px', color: 'var(--ink)' }}>{item.title}</div>
        <div style={{ marginTop: 3, font: '600 12px var(--font-ui)', color: 'var(--slate-500)' }}>{creator.short} · {item.slides} slides · {item.posted}</div>
        <div style={{ marginTop: 6, display: 'inline-flex', alignItems: 'center', gap: 5, font: '700 11px var(--font-ui)', color: 'var(--blue-700)' }}>
          <Icon name="music-2" size={12} />{item.marked}
        </div>
      </div>
      {approved
        ? <span style={{ display: 'inline-flex', alignItems: 'center', gap: 5, padding: '8px 12px', borderRadius: 999, background: 'var(--green-soft)', color: 'var(--green)', font: '700 13px var(--font-ui)', flex: '0 0 auto' }}><Icon name="check" size={14} />Approved</span>
        : <Icon name="chevron-right" size={18} color="var(--slate-300)" style={{ flex: '0 0 auto' }} />}
    </Card>
  );
}

function AccountRow({ item, creator, onClick }) {
  const { Icon } = window.NoniDesignSystem_710e43;
  const sentBack = item.state === 'needs_changes';
  return (
    <Card pad={12} onClick={onClick} style={{ display: 'flex', gap: 12, alignItems: 'center' }}>
      <Avatar name={creator.name} size={40} tone={sentBack ? 'quiet' : 'brand'} />
      <div style={{ flex: 1, minWidth: 0 }}>
        <div style={{ display: 'flex', alignItems: 'center', gap: 7 }}>
          <span style={{ font: '700 15px var(--font-display)', letterSpacing: '-0.2px', color: 'var(--ink)' }}>{creator.name}</span>
          <TypeChip tone={sentBack ? 'warn' : 'quiet'}>{sentBack ? 'Needs changes' : 'Pending'}</TypeChip>
        </div>
        <div style={{ marginTop: 3, font: '600 12px var(--font-ui)', color: 'var(--slate-500)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>@{item.tiktok} · @{item.instagram}</div>
        <div style={{ marginTop: 2, font: '400 12px var(--font-ui)', color: 'var(--slate-400)' }}>{sentBack ? item.reason : item.submitted}</div>
      </div>
      <Icon name="chevron-right" size={18} color="var(--slate-300)" />
    </Card>
  );
}

Object.assign(window, { ReviewScreen, SubmissionRow, MusicApprovalRow, AccountRow });
