  { id: 'cm2', name: 'Sofia Marek', role: 'Campaign manager', meta: 'Runs the Tuesday and Thursday briefs', you: false },
  { id: 'cm1', name: 'You', role: 'Campaign manager', meta: 'Signed in on this phone', you: true },
];

const SEED_CHANNELS = [
  { id: 'ch1', name: 'general', last: 'Elan: Budget tops up Friday — keep approvals moving.', unread: 2, members: 3 },
  { id: 'ch2', name: 'wins', last: 'Sofia: Fabri’s tripod reel just crossed 400k 🎉', unread: 0, members: 3 },
  { id: 'ch3', name: 'briefs', last: 'You: Next week leans slideshows, drafts up tonight.', unread: 0, members: 2 },
];

function CreatorsScreen({ state = 'default', onOpen, onOpenMember, onOpenChannel }) {
  const { Icon, EmptyState } = window.NoniDesignSystem_710e43;
  const A = window.NONI_ADMIN;
  const [lane, setLane] = React.useState(0);
  const [sort, setSort] = React.useState(0);
  const [channels, setChannels] = React.useState(SEED_CHANNELS);
  const [creating, setCreating] = React.useState(false);
  const [chName, setChName] = React.useState('');
  const loading = state === 'loading';
  const empty = state === 'empty';
  const list = empty ? [] : A.creators.filter((c) => c.status === 'approved');

  const addChannel = () => {
    const n = chName.trim().toLowerCase().replace(/\s+/g, '-').replace(/[^a-z0-9-]/g, '');
    if (!n) return;
    setChannels([...channels, { id: 'ch' + (channels.length + 1) + n, name: n, last: 'You created this channel.', unread: 0, members: 1 }]);
    setChName(''); setCreating(false);
  };

  return (
    <AdminScreen>
      <AdminHeader title="Team" subtitle={loading ? null : `${list.length} creators · ${TEAM_PEOPLE.filter((p) => !p.you && p.role !== 'Admin').length + 1} campaign managers · 1 admin`} />
      {!loading && !empty ? <Segmented items={['Creators', 'Team', 'Channels']} active={lane} onSelect={setLane} /> : null}

      {loading ? <SkeletonRows n={3} height={112} /> : null}
      {!loading && empty ? <EmptyState icon="users" title="No creators yet" body="Invite from Settings. They upload warm-up proof, you approve, then briefs start landing." style={{ marginTop: 40 }} /> : null}

      {!loading && !empty && lane === 0 ? (
        <>
        <div style={{ display: 'flex', gap: 6 }}>
          {['Earnings', 'Views', 'Posts'].map((s, i) => (
            <button key={s} type="button" onClick={() => setSort(i)}
              style={{ flex: '0 0 auto', display: 'inline-flex', alignItems: 'center', gap: 5, padding: '7px 12px', borderRadius: 999, border: 'none', cursor: 'pointer', background: sort === i ? 'var(--blue-100)' : 'var(--white)', boxShadow: sort === i ? 'none' : 'var(--shadow-card)', color: sort === i ? 'var(--blue-700)' : 'var(--slate-500)', font: '700 12px var(--font-ui)' }}>
              {sort === i ? <Icon name="trending-up" size={12} /> : null}{s}
            </button>
          ))}
        </div>
        <div style={{ display: 'flex', flexDirection: 'column', gap: 10 }}>
          {list.map((c) => (
            <Card key={c.id} pad={14} onClick={() => onOpen && onOpen(c)}>
              <div style={{ display: 'flex', alignItems: 'center', gap: 12 }}>
                <Avatar name={c.name} size={44} />
                <div style={{ flex: 1, minWidth: 0 }}>
                  <div style={{ font: '700 17px var(--font-display)', letterSpacing: '-0.3px', color: 'var(--ink)' }}>{c.name}</div>
                  <div style={{ marginTop: 2, font: '600 12px var(--font-ui)', color: 'var(--slate-400)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>@{c.tiktok}</div>
                </div>
                <Icon name="chevron-right" size={18} color="var(--slate-300)" />
              </div>
              <div style={{ marginTop: 13, display: 'flex', gap: 8 }}>
                {[[c.earned, 'earned'], [c.posts, 'posts'], [c.views, 'views']].map(([v, l]) => (
                  <div key={l} style={{ flex: 1, padding: '10px 11px', borderRadius: 'var(--radius-sm)', background: 'var(--off-white)' }}>
                    <div style={{ font: '700 16px var(--font-display)', letterSpacing: '-0.3px', color: 'var(--ink)' }}>{v}</div>
                    <div style={{ marginTop: 1, font: '600 11px var(--font-ui)', color: 'var(--slate-400)' }}>{l}</div>
                  </div>
                ))}
              </div>
            </Card>
          ))}
        </div>
        </>
      ) : null}

      {!loading && !empty && lane === 1 ? (
        <div style={{ display: 'flex', flexDirection: 'column', gap: 10 }}>
          {TEAM_PEOPLE.map((p) => (
            <Card key={p.id} pad={14} onClick={!p.you ? () => onOpenMember && onOpenMember(p) : undefined} style={p.you ? { opacity: 0.75 } : undefined}>
              <div style={{ display: 'flex', alignItems: 'center', gap: 12 }}>
                <Avatar name={p.you ? 'You' : p.name} size={44} tone={p.role === 'Admin' ? undefined : 'quiet'} />
                <div style={{ flex: 1, minWidth: 0 }}>
                  <div style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
                    <span style={{ font: '700 17px var(--font-display)', letterSpacing: '-0.3px', color: 'var(--ink)' }}>{p.name}</span>
                    <TypeChip tone={p.role === 'Admin' ? 'brand' : undefined}>{p.role}</TypeChip>
                  </div>
                  <div style={{ marginTop: 2, font: '600 12px var(--font-ui)', color: 'var(--slate-400)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{p.meta}</div>
                </div>
                {!p.you ? (
                  <span style={{ width: 36, height: 36, flex: '0 0 auto', borderRadius: 999, background: 'var(--blue-100)', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}>
                    <Icon name="message-circle" size={17} color="var(--blue-700)" />

/* Creator-scoped week: stepper (past/next week), only this creator's briefs.
   Green check = posted, orange dot = not yet. Right label = Post | Slideshow. */
function CreatorWeek({ creator }) {
  const { Icon } = window.NoniDesignSystem_710e43;
  const A = window.NONI_ADMIN;
  const [w, setW] = React.useState(0);
  const mine = A.calendar.map((d) => ({ day: d.day, items: d.items.filter((it) => it.c === creator.short) }));
  const weeks = {
    '-1': { label: 'Aug 3 – 9', days: [
      { day: 'Mon 3', items: [{ t: 'First touch warm-up', f: 'video', s: 'posted' }] },
      { day: 'Tue 4', items: [] },
      { day: 'Wed 5', items: [{ t: 'Coach reply window', f: 'photo_carousel', s: 'posted' }] },
      { day: 'Thu 6', items: [] },
      { day: 'Fri 7', items: [{ t: 'Match film breakdown', f: 'video', s: 'posted' }] },
      { day: 'Sat 8', items: [{ t: '3 stats that decide Sunday', f: 'photo_carousel', s: 'assigned' }] },
      { day: 'Sun 9', items: [] },
    ] },
    '0': { label: 'Aug 10 – 16', days: mine },
    '1': { label: 'Aug 17 – 23', days: [
      { day: 'Mon 17', items: [{ t: 'Preseason check-in', f: 'video', s: 'assigned' }] },
      { day: 'Tue 18', items: [] },
      { day: 'Wed 19', items: [{ t: 'Drill of the week', f: 'photo_carousel', s: 'assigned' }] },
      { day: 'Thu 20', items: [] },
      { day: 'Fri 21', items: [{ t: 'Scrimmage highlights', f: 'video', s: 'assigned' }] },
      { day: 'Sat 22', items: [] },
      { day: 'Sun 23', items: [] },
    ] },
  };
  const wk = weeks['' + w];
  const posted = (s) => s === 'posted' || s === 'approved';
  const all = wk.days.flatMap((d) => d.items);
  const nDone = all.filter((it) => posted(it.s)).length;
  const step = (dir, disabled) => (
    <button type="button" onClick={() => setW(w + dir)} disabled={disabled} aria-label={dir < 0 ? 'Previous week' : 'Next week'}
      style={{ width: 32, height: 32, flex: '0 0 auto', borderRadius: 999, border: 'none', cursor: disabled ? 'default' : 'pointer', opacity: disabled ? 0.35 : 1, background: 'var(--white)', boxShadow: 'var(--shadow-card)', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}>
      <Icon name={dir < 0 ? 'chevron-left' : 'chevron-right'} size={16} color="var(--ink)" />
    </button>
  );
  return (
    <div style={{ display: 'flex', flexDirection: 'column', gap: 10 }}>
      <div style={{ display: 'flex', alignItems: 'center', gap: 10 }}>
        {step(-1, w === -1)}
        <div style={{ flex: 1, textAlign: 'center' }}>
          <div style={{ font: '700 14.5px var(--font-display)', letterSpacing: '-0.2px', color: 'var(--ink)' }}>{wk.label}</div>
          <div style={{ marginTop: 1, font: '600 11.5px var(--font-ui)', color: 'var(--slate-400)' }}>{all.length ? `${nDone} of ${all.length} posted` : 'No brief this week'}</div>
        </div>
        {step(1, w === 1)}
      </div>
      {wk.days.map((d) => (
        <Card key={d.day} pad={12} style={{ display: 'flex', gap: 12, alignItems: d.items.length ? 'flex-start' : 'center' }}>
          <div style={{ width: 46, flex: '0 0 auto' }}>
            <div style={{ font: '700 11px var(--font-ui)', letterSpacing: '0.4px', textTransform: 'uppercase', color: 'var(--slate-400)' }}>{d.day.split(' ')[0]}</div>
            <div style={{ font: '700 20px var(--font-display)', letterSpacing: '-0.4px', color: 'var(--ink)' }}>{d.day.split(' ')[1]}</div>
          </div>
          {d.items.length ? (
            <div style={{ flex: 1, minWidth: 0, display: 'flex', flexDirection: 'column', gap: 6 }}>
              {d.items.map((it, i) => (
                <div key={i} style={{ display: 'flex', alignItems: 'center', gap: 8, padding: '8px 10px', borderRadius: 'var(--radius-sm)', background: 'var(--off-white)' }}>
                  {posted(it.s)
                    ? <Icon name="circle-check-big" size={14} color="var(--green)" style={{ flex: '0 0 auto' }} />
                    : <span style={{ width: 8, height: 8, flex: '0 0 auto', borderRadius: 999, background: 'var(--amber)' }}></span>}
                  <span style={{ flex: 1, minWidth: 0, font: '600 13px var(--font-ui)', color: 'var(--ink)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{it.t}</span>
                  <Icon name={window.isReel(it.f) ? 'video' : 'images'} size={13} color="var(--slate-400)" />
                  <span style={{ font: '600 11px var(--font-ui)', color: 'var(--slate-400)' }}>{window.isReel(it.f) ? 'Post' : 'Slideshow'}</span>
                </div>
              ))}
            </div>
          ) : (
            <span style={{ flex: 1, font: '400 13px var(--font-ui)', color: 'var(--slate-300)' }}>Rest day</span>
          )}
        </Card>
      ))}
    </div>
  );
}

/* Per-creator analytics: range · views chart · platform split. Reuses the
   Analytics tab's chart engine, scoped to this creator. */
function CreatorAnalytics({ creator }) {
  const { MAreaChart, MMenu, MItem, aRangeData, A_WEEKLY, A_RANGES, Icon } = { ...window, Icon: window.NoniDesignSystem_710e43.Icon };
  const [range, setRange] = React.useState('Last month');
  const n = /m/i.test(creator.views) ? parseFloat(creator.views) * 1000 : parseFloat(creator.views);
  const share = Math.min(1, (n || 400) / 4200);
  const chart = aRangeData(range, A_WEEKLY.map((v) => +(v * share).toFixed(1)));
  const tt = 0.76;
  return (
    <Card pad={14}>
      <div style={{ display: 'flex', alignItems: 'center', gap: 8, marginBottom: 12 }}>
        <SectionLabel style={{ padding: 0, flex: 1 }}>Views · {range.toLowerCase()}</SectionLabel>
        <MMenu trigger={range}>
          {(close) => A_RANGES.map((r) => <MItem key={r} label={r} on={range === r} pick={() => close(() => setRange(r))} />)}
        </MMenu>
      </div>
      <MAreaChart series={chart.data} labels={chart.labels} />
      <div style={{ marginTop: 14, display: 'flex', flexDirection: 'column', gap: 11 }}>
        {[['TikTok', 'music-2', tt], ['Instagram', 'at-sign', 1 - tt]].map(([l, icon, f]) => (
          <div key={l} style={{ display: 'flex', alignItems: 'center', gap: 9 }}>
            <Icon name={icon} size={13} color="var(--slate-500)" />
            <span style={{ width: 72, font: '600 12.5px var(--font-ui)', color: 'var(--slate-500)' }}>{l}</span>
            <span style={{ flex: 1, height: 8, borderRadius: 999, background: 'var(--fill-quiet)', overflow: 'hidden' }}>
              <span style={{ display: 'block', width: Math.round(f * 100) + '%', height: '100%', borderRadius: 999, background: 'var(--blue-500)' }}></span>
            </span>
            <span style={{ width: 34, textAlign: 'right', font: '700 12.5px var(--font-ui)', color: 'var(--ink)' }}>{Math.round(f * 100)}%</span>
          </div>
        ))}
      </div>
    </Card>
  );
}

                  ) : <Icon name="images" size={22} color="var(--blue-300)" />}

            <Icon name="send" size={18} color="var(--white)" />
          </button>
        </div>
      </div>
    </div>
  );
}
