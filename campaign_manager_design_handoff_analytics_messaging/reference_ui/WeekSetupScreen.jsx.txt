/* Week setup — one question, once a week: how many videos and slideshows a
   day. Rows are created from that. Post types are not picked here; each new
   post suggests its kind in the editor, weighted by what the week still lacks. */

/* Scale the default type weights to a weekly total, largest remainder so
   the split always adds up. */
function autoSplit(types, total) {
  const w = types.reduce((a, t) => a + t.d, 0);
  const raw = types.map((t) => (t.d / w) * total);
  const base = raw.map(Math.floor);
  let left = total - base.reduce((a, b) => a + b, 0);
  raw.map((r, i) => [r - base[i], i]).sort((a, b) => b[0] - a[0]).slice(0, left).forEach(([, i]) => { base[i] += 1; });
  return Object.fromEntries(types.map((t, i) => [t.key, base[i]]));
}

function WeekSetup({ step: step0 = 0, state = 'default', onDone, onBack, title = 'Week 15 · Aug 17–23' }) {
  const { Button, Icon } = window.NoniDesignSystem_710e43;
  const [vDay, setVDay] = React.useState(2);
  const [sDay, setSDay] = React.useState(1);
  const videos = vDay * 7, slideshows = sDay * 7;
  /* Types are not set here any more: each new post suggests its kind in the
     editor, weighted by what the week still needs. */
  const total = videos + slideshows;

  return (
    <AdminScreen bottom={110} pad={false} footer={
      <ActionBar>
        <Button variant="primary" size="md" block disabled={total === 0} onClick={() => onDone && onDone()} style={{ flex: 1 }}>
          {`Create ${total} rows`}
        </Button>
      </ActionBar>
    }>
      <div style={{ flex: 1, minHeight: 0, overflowY: 'auto', display: 'flex', flexDirection: 'column', gap: 16, padding: `0 0 110px` }}>
        <PushHeader title={title} meta="Week setup" onBack={onBack} />
        <div style={{ padding: `0 ${window.GUT}px`, display: 'flex', flexDirection: 'column', gap: 16 }}>
          <div>
            <h1 style={{ margin: 0, font: '700 28px var(--font-display)', letterSpacing: 'var(--tracking-title)', color: 'var(--ink)' }}>Posts a day</h1>
          </div>

          <div style={{ display: 'flex', flexDirection: 'column', gap: 12 }}>
              <p style={{ margin: '-6px 0 0', font: '400 14px/1.45 var(--font-ui)', color: 'var(--slate-500)' }}>How many should each creator post a day? That is the whole setup.</p>
              <RatioCard label="Videos a day" hint="Reels" value={vDay} onChange={setVDay} icon="video" />
              <RatioCard label="Slideshows a day" hint="Photo carousels" value={sDay} onChange={setSDay} icon="images" />
              <div style={{ display: 'flex', alignItems: 'center', gap: 10, padding: '12px 14px', borderRadius: 'var(--radius-md)', background: 'var(--blue-50)' }}>
                <Icon name="sparkles" size={16} color="var(--blue-600)" />
                <span style={{ flex: 1, font: '600 13.5px/1.4 var(--font-ui)', color: 'var(--ink)' }}>
                  {vDay + sDay} posts a day, {total} this week: {videos} videos and {slideshows} slideshows. Each post suggests its kind when you open it.
                </span>
              </div>
            </div>
        </div>
      </div>
    </AdminScreen>
  );
}

function RatioCard({ label, hint, value, onChange, icon }) {
  const { Icon } = window.NoniDesignSystem_710e43;
  const step = (d) => onChange(Math.max(0, value + d));
  return (
    <Card pad={16} style={{ display: 'flex', alignItems: 'center', gap: 14 }}>
      <span style={{ width: 42, height: 42, flex: '0 0 auto', borderRadius: 999, background: 'var(--blue-100)', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}>
        <Icon name={icon} size={19} color="var(--blue-700)" />
      </span>
      <div style={{ flex: 1, minWidth: 0 }}>
        <div style={{ font: '700 17px var(--font-display)', letterSpacing: '-0.3px', color: 'var(--ink)' }}>{label}</div>
        <div style={{ marginTop: 2, font: '600 13px var(--font-ui)', color: 'var(--slate-400)' }}>{hint}</div>
      </div>
      <StepControl value={value} onStep={step} />
    </Card>
  );
}

function StepControl({ value, onStep, w = 34 }) {
  const { Icon } = window.NoniDesignSystem_710e43;
  const btn = { width: w, height: w, borderRadius: 999, border: 'none', cursor: 'pointer', background: 'var(--fill-quiet)', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' };
  return (
    <div style={{ display: 'flex', alignItems: 'center', gap: 9 }}>
      <button type="button" onClick={() => onStep(-1)} aria-label="Decrease" style={btn}>
        <span style={{ width: 12, height: 2, borderRadius: 2, background: 'var(--slate-500)' }} />
      </button>
      <span style={{ minWidth: 28, textAlign: 'center', font: '700 22px var(--font-display)', letterSpacing: '-0.5px', color: 'var(--ink)' }}>{value}</span>
      <button type="button" onClick={() => onStep(1)} aria-label="Increase" style={btn}>
        <Icon name="plus" size={16} color="var(--slate-500)" />
      </button>
    </div>
  );
}

function SplitList({ types, split, onChange, sum, target, noun }) {
  const { Icon } = window.NoniDesignSystem_710e43;
  const diff = sum - target;
  const good = diff === 0;
  return (
    <div style={{ display: 'flex', flexDirection: 'column', gap: 10 }}>
      <div style={{ display: 'flex', alignItems: 'center', gap: 9, padding: '12px 14px', borderRadius: 'var(--radius-md)', background: good ? 'var(--green-soft)' : 'var(--amber-soft)' }}>
        <Icon name={good ? 'circle-check-big' : 'circle-alert'} size={17} color={good ? 'var(--green)' : 'var(--amber)'} />
        <span style={{ flex: 1, font: '700 14px var(--font-ui)', color: good ? 'var(--green)' : '#8A5A0E' }}>
          {good ? `${sum} of ${target} ${noun} assigned` : diff > 0 ? `${diff} over. Take ${diff} off a type.` : `${-diff} left to assign.`}
        </span>
      </div>
      {types.map((t) => (
        <Card key={t.key} pad={13} style={{ display: 'flex', alignItems: 'center', gap: 12 }}>
          <div style={{ flex: 1, minWidth: 0 }}>
            <div style={{ font: '700 15px var(--font-display)', letterSpacing: '-0.2px', color: 'var(--ink)' }}>{t.label}</div>
            <div style={{ marginTop: 2, font: '600 12px var(--font-ui)', color: 'var(--slate-400)' }}>{t.hint}</div>
          </div>
          <StepControl value={split[t.key]} onStep={(d) => onChange(t.key, Math.max(0, split[t.key] + d))} w={30} />
        </Card>
      ))}
    </div>
  );
}

Object.assign(window, { WeekSetup, RatioCard, StepControl, SplitList, autoSplit, VIDEO_TYPES, SLIDE_TYPES });
