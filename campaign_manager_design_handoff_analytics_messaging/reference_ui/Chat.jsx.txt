/* Messaging for campaign managers. One group chat per brief plus direct
   messages, reached from the MsgButton on every Briefs surface. Bubbles
   support replies, forwards, voice notes and uploads. */

const CHAT_NS = () => window.NoniDesignSystem_710e43;

/* --- shared chat vocabulary --- */

function ChatBubble({ who, time, me, quote, forward, children, reactions }) {
  const { Icon } = CHAT_NS();
  return (
    <div style={{ display: 'flex', gap: 9, flexDirection: me ? 'row-reverse' : 'row', alignItems: 'flex-end' }}>
      {!me ? <Avatar name={who} size={28} tone="quiet" /> : null}
      <div style={{ display: 'flex', flexDirection: 'column', gap: 3, alignItems: me ? 'flex-end' : 'flex-start', maxWidth: '82%' }}>
        <span style={{ font: '600 11.5px var(--font-ui)', color: 'var(--slate-400)', padding: '0 4px', whiteSpace: 'nowrap' }}>{me ? time : `${who} · ${time}`}</span>
        <div style={{ padding: '10px 13px', borderRadius: 18, borderBottomRightRadius: me ? 6 : 18, borderBottomLeftRadius: me ? 18 : 6, background: me ? 'var(--blue-500)' : 'var(--white)', color: me ? 'var(--white)' : 'var(--ink)', boxShadow: me ? 'var(--shadow-accent)' : 'var(--shadow-card)', font: '400 14.5px/1.45 var(--font-ui)', display: 'flex', flexDirection: 'column', gap: 8 }}>
          {forward ? (
            <span style={{ display: 'inline-flex', alignItems: 'center', gap: 5, font: '600 11.5px var(--font-ui)', color: me ? 'rgba(255,255,255,0.75)' : 'var(--slate-400)' }}>
              <Icon name="share-2" size={12} color={me ? 'rgba(255,255,255,0.75)' : 'var(--slate-400)'} />{forward}
            </span>
          ) : null}
          {quote ? (
            <div style={{ display: 'flex', gap: 8, padding: '7px 10px', borderRadius: 10, background: me ? 'rgba(255,255,255,0.16)' : 'var(--fill-quiet)' }}>
              <span style={{ width: 3, flex: '0 0 auto', borderRadius: 999, background: me ? 'rgba(255,255,255,0.6)' : 'var(--blue-300)' }} />
              <span style={{ minWidth: 0 }}>
                <span style={{ display: 'block', font: '700 11.5px var(--font-ui)', color: me ? 'var(--white)' : 'var(--blue-700)' }}>{quote[0]}</span>
                <span style={{ display: 'block', marginTop: 1, font: '400 12.5px/1.35 var(--font-ui)', color: me ? 'rgba(255,255,255,0.8)' : 'var(--slate-500)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis', maxWidth: 210 }}>{quote[1]}</span>
              </span>
            </div>
          ) : null}
          {children}
        </div>
        {reactions ? (
          <div style={{ display: 'flex', gap: 4, marginTop: -1, padding: '0 4px' }}>
            {reactions.map(([icon, n]) => (
              <span key={icon} style={{ display: 'inline-flex', alignItems: 'center', gap: 4, padding: '3px 8px', borderRadius: 999, background: 'var(--white)', boxShadow: 'var(--shadow-card)', font: '700 11px var(--font-ui)', color: 'var(--slate-500)' }}>
                <Icon name={icon} size={11} color="var(--blue-600)" />{n}
              </span>
            ))}
          </div>
        ) : null}
      </div>
    </div>
  );
}

function VoiceNote({ me, duration }) {
  const { Icon } = CHAT_NS();
  const bars = [6, 12, 9, 16, 11, 18, 8, 14, 10, 16, 7, 13, 9, 15, 6, 11, 8, 13];
  const fg = me ? 'var(--white)' : 'var(--blue-600)';
  return (
    <span style={{ display: 'inline-flex', alignItems: 'center', gap: 10 }}>
      <span style={{ width: 30, height: 30, flex: '0 0 auto', borderRadius: 999, background: me ? 'rgba(255,255,255,0.22)' : 'var(--blue-100)', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}>
        <Icon name="play" size={13} color={fg} />
      </span>
      <span style={{ display: 'inline-flex', alignItems: 'center', gap: 2 }}>
        {bars.map((h, i) => <span key={i} style={{ width: 3, height: h, borderRadius: 999, background: me ? (i < 8 ? 'var(--white)' : 'rgba(255,255,255,0.4)') : (i < 8 ? 'var(--blue-500)' : 'var(--blue-100)') }} />)}
      </span>
      <span style={{ font: '700 12px var(--font-ui)', color: me ? 'rgba(255,255,255,0.85)' : 'var(--slate-500)' }}>{duration}</span>
    </span>
  );
}

function UploadThumbs({ n = 2, caption }) {
  const { Icon } = CHAT_NS();
  return (
    <span style={{ display: 'flex', flexDirection: 'column', gap: 7 }}>
      <span style={{ display: 'flex', gap: 6 }}>
        {Array.from({ length: n }).map((_, i) => (
          <span key={i} style={{ width: 96, height: 120, borderRadius: 12, background: 'linear-gradient(160deg,#E7F4FD 0%,#DCE7F0 100%)', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}>
            <Icon name="images" size={22} color="var(--blue-300)" />
          </span>
        ))}
      </span>
      {caption ? <span>{caption}</span> : null}
    </span>
  );
}

function PostRef({ me, label }) {
  const { Icon } = CHAT_NS();
  return (
    <span style={{ display: 'flex', alignItems: 'center', gap: 9, padding: '9px 11px', borderRadius: 12, background: me ? 'rgba(255,255,255,0.16)' : 'var(--fill-quiet)' }}>
      <Icon name="video" size={14} color={me ? 'var(--white)' : 'var(--blue-700)'} />
      <span style={{ flex: 1, font: '700 12.5px var(--font-ui)', color: me ? 'var(--white)' : 'var(--ink)' }}>{label}</span>
      <Icon name="chevron-right" size={13} color={me ? 'rgba(255,255,255,0.7)' : 'var(--slate-400)'} />
    </span>
  );
}

function Composer({ placeholder }) {
  const { Icon } = CHAT_NS();
  const round = (bg, shadow) => ({ width: 42, height: 42, flex: '0 0 auto', borderRadius: 999, border: 'none', cursor: 'pointer', background: bg, boxShadow: shadow || 'none', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' });
  return (
    <div style={{ position: 'absolute', bottom: 0, left: 0, right: 0, padding: '12px 16px 24px', display: 'flex', alignItems: 'center', gap: 8, background: 'var(--off-white)' }}>
      <button type="button" aria-label="Add attachment" style={round('var(--white)', 'var(--shadow-card)')}>
        <Icon name="plus" size={18} color="var(--slate-500)" />
      </button>
      <div style={{ flex: 1, height: 44, minWidth: 0, borderRadius: 999, background: 'var(--white)', boxShadow: 'var(--shadow-card)', display: 'flex', alignItems: 'center', gap: 8, padding: '0 6px 0 16px' }}>
        <span style={{ flex: 1, minWidth: 0, font: '400 14px var(--font-ui)', color: 'var(--slate-400)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{placeholder}</span>
        <button type="button" aria-label="Record voice note" style={{ width: 34, height: 34, flex: '0 0 auto', borderRadius: 999, border: 'none', cursor: 'pointer', background: 'var(--fill-quiet)', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}>
          <Icon name="mic" size={15} color="var(--slate-500)" />
        </button>
      </div>
      <button type="button" aria-label="Send" style={round('var(--blue-500)', 'var(--shadow-accent)')}>
        <Icon name="send" size={17} color="var(--white)" />
      </button>
    </div>
  );
}

function ChatScreen({ title, meta, right, placeholder, onBack, children }) {
  return (
    <AdminScreen pad={false} bottom={0} footer={<Composer placeholder={placeholder} />}>
      <div style={{ flex: 1, minHeight: 0, overflowY: 'auto', display: 'flex', flexDirection: 'column' }}>
        <PushHeader title={title} meta={meta} onBack={onBack} right={right} />
        <div style={{ padding: `10px ${window.GUT}px 100px`, display: 'flex', flexDirection: 'column', gap: 15 }}>{children}</div>
      </div>
    </AdminScreen>
  );
}

const ChatDivider = ({ children }) => (
