/* First run for a brand-new campaign manager account (no week yet, Week 1),
   plus the multi-manager collaboration surfaces: every manager sees the same
   weeks, presence shows who is filling what, and each week has one shared
   thread. */

const MANAGERS = [
  { name: 'You', me: true },
  { name: 'Maya' },
  { name: 'Jordan' },
];

function AvatarStack({ people, size = 30 }) {
  return (
    <span style={{ display: 'inline-flex', alignItems: 'center' }}>
      {people.map((p, i) => (
        <span key={p.name} style={{ marginLeft: i ? -8 : 0, borderRadius: 999, boxShadow: '0 0 0 2px var(--white)', display: 'inline-flex' }}>
          <Avatar name={p.name} size={size} tone={p.me ? 'brand' : 'quiet'} />
        </span>
      ))}
    </span>
  );
}

/* 03 — the Briefs tab as a list of week cards: Next week always on top, then
   every week before it, most recent to oldest. Each card clicks into that
   week. Status dot: orange while in progress, green once complete. */
function SharedWeekBriefs({ onOpenWeek }) {
  const { Icon } = window.NoniDesignSystem_710e43;
  const weeks = [
    { label: 'Week 2', range: 'Aug 24 to 30', status: 'next' },
    { label: 'Week 1', range: 'Aug 17 to 23', status: 'done', complete: true, stats: [['52K', 'views/day'], ['$1,240', 'sales'], ['30', 'posts']] },
  ];
  return (
    <AdminScreen gap={10}>
      <AdminHeader title="Briefs" subtitle="A week is one shared pool of posts for the whole roster." right={<MsgButton count={3} />} />
      {weeks.map((w) => (
        <Card key={w.label} pad={16} onClick={() => onOpenWeek && onOpenWeek(w)} style={{ display: 'flex', flexDirection: 'column', gap: 14, cursor: 'pointer' }}>
          <div style={{ display: 'flex', alignItems: 'center', gap: 10 }}>
            <div style={{ flex: 1, minWidth: 0 }}>
              <div style={{ font: '700 19px var(--font-display)', letterSpacing: '-0.3px', color: 'var(--ink)' }}>{w.label}</div>
              <div style={{ marginTop: 2, display: 'flex', alignItems: 'center', gap: 7, font: '600 13px var(--font-ui)', color: 'var(--slate-400)' }}>
                {w.status !== 'next' ? <span style={{ width: 8, height: 8, flex: '0 0 auto', borderRadius: 999, background: w.complete ? 'var(--green)' : 'var(--amber)' }} /> : null}
                {w.range} · {w.status === 'next' ? 'opens Sunday' : w.complete ? 'complete' : 'in progress'}
              </div>
            </div>
            {w.status === 'next' ? <TypeChip tone="brand">Next week</TypeChip> : <AvatarStack people={MANAGERS} />}
            <Icon name="chevron-right" size={16} color="var(--slate-300)" style={{ flex: '0 0 auto' }} />
          </div>
          {w.stats ? (
            <div style={{ display: 'flex', gap: 6 }}>
              {w.stats.map(([v, l]) => (
                <span key={l} style={{ display: 'inline-flex', alignItems: 'center', gap: 5, padding: '6px 10px', borderRadius: 999, background: 'var(--off-white)', font: '600 11.5px var(--font-ui)', color: 'var(--slate-500)' }}>
                  <span style={{ font: '700 12px var(--font-ui)', color: 'var(--ink)' }}>{v}</span>{l}
                </span>
              ))}
            </div>
          ) : w.lanes ? (
            <div style={{ display: 'flex', gap: 10 }}>
              {w.lanes.map(([icon, label, done, total]) => (
                <div key={label} style={{ flex: 1, padding: '12px 14px', borderRadius: 'var(--radius-md)', background: 'var(--fill-quiet)', display: 'flex', flexDirection: 'column', gap: 4 }}>
                  <span style={{ display: 'inline-flex', alignItems: 'center', gap: 6, font: '700 12px var(--font-ui)', color: 'var(--slate-500)' }}>
                    <Icon name={icon} size={13} color="var(--slate-500)" />{label}
                  </span>
                  <span style={{ font: '700 22px var(--font-display)', letterSpacing: '-0.5px', color: 'var(--ink)' }}>{done}<span style={{ font: '700 14px var(--font-display)', color: 'var(--slate-400)' }}> / {total}</span></span>
                </div>
              ))}
            </div>
          ) : (
            <p style={{ margin: 0, font: '400 12.5px/1.4 var(--font-ui)', color: 'var(--slate-400)' }}>Not planned yet. Opens Sunday, tap to start it.</p>
          )}
        </Card>
      ))}
    </AdminScreen>
  );
}

    <AdminScreen pad={false}>
      <PushHeader title="Week 1" meta="Aug 17 to 23 · complete" onBack={onBack} right={<MsgButton count={3} />} />
      <div style={{ flex: 1, minHeight: 0, overflowY: 'auto', padding: `2px ${window.GUT}px 40px`, display: 'flex', flexDirection: 'column', gap: 12 }}>
        <div style={{ display: 'flex', gap: 8, flex: '0 0 auto' }}>
          {[['52K', 'Views/day'], ['$1,240', 'Sales'], ['30', 'Posts']].map(([v, l]) => (
            <Card key={l} pad={12} style={{ flex: 1 }}>
              <div style={{ font: '700 20px var(--font-display)', letterSpacing: '-0.4px', color: l === 'Sales' ? 'var(--green)' : 'var(--ink)' }}>{v}</div>
              <div style={{ marginTop: 2, font: '700 11px var(--font-ui)', letterSpacing: '0.3px', textTransform: 'uppercase', color: 'var(--slate-400)' }}>{l}</div>
