/* Library — compact header, three lanes: Posts (best work grouped by
   creator), References, Ideas. Ideas is one-line capture: type, pick video
   or slideshow, save — AI cleans it into a draft. */

const libViewsN = (v) => /m/i.test(v) ? parseFloat(v) * 1e6 : /k/i.test(v) ? parseFloat(v) * 1e3 : parseFloat(v) || 0;
const libClean = (t) => { const s = t.trim().replace(/\s+/g, ' '); return s.charAt(0).toUpperCase() + s.slice(1) + (/[.!?]$/.test(s) ? '' : '.'); };

function LibraryScreen({ state = 'default', chip: chip0 = 0, capture = '' }) {
  const { Icon, EmptyState, Button } = window.NoniDesignSystem_710e43;
  const L = window.NONI_ADMIN.library;
  const [chip, setChip] = React.useState(chip0);
  const [text, setText] = React.useState(capture);
  const [fmt, setFmt] = React.useState('video');
  const [ideas, setIdeas] = React.useState(L.ideas);
  const loading = state === 'loading';
  const empty = state === 'empty';
  const chips = ['Posts', 'References', 'Ideas'];

  const save = () => {
    if (!text.trim()) return;
    setIdeas([{ id: 'new' + Date.now(), body: libClean(text), age: 'Just now', format: fmt, drafted: true }, ...ideas]);
    setText('');
  };

  const groups = (() => {
    const by = {};
    L.ours.forEach((it) => { (by[it.creator] = by[it.creator] || []).push(it); });
    return Object.entries(by)
      .map(([name, items]) => ({ name, items: items.sort((a, b) => libViewsN(b.views) - libViewsN(a.views)), total: items.reduce((n, i) => n + libViewsN(i.views), 0) }))
      .sort((a, b) => b.total - a.total);
  })();

  return (
    <AdminScreen gap={12}>
      <div style={{ display: 'flex', alignItems: 'center', gap: 8, padding: '4px 0 0' }}>
        <h1 style={{ margin: 0, font: '700 22px var(--font-display)', letterSpacing: '-0.5px', color: 'var(--ink)' }}>Library</h1>
        <span style={{ flex: 1 }}></span>
        <div style={{ display: 'flex', gap: 3, padding: 3, borderRadius: 999, background: 'var(--fill-quiet)' }}>
          {chips.map((c, i) => (
            <button key={c} type="button" onClick={() => setChip(i)} aria-pressed={chip === i}
              style={{ padding: '7px 11px', borderRadius: 999, border: 'none', cursor: 'pointer', background: chip === i ? 'var(--white)' : 'transparent', boxShadow: chip === i ? 'var(--shadow-card)' : 'none', color: chip === i ? 'var(--ink)' : 'var(--slate-400)', font: '700 12.5px var(--font-ui)', whiteSpace: 'nowrap' }}>{c}</button>
          ))}
        </div>
      </div>

      {loading ? <SkeletonRows n={4} height={76} /> : null}

      {!loading && empty ? (
        <EmptyState icon="images"
          title={chip === 0 ? 'No posts yet' : chip === 1 ? 'No references yet' : 'No ideas saved yet'}
          body={chip === 0 ? 'Posts land here the day they go live, sorted by performance.' : chip === 1 ? 'Paste a TikTok or Instagram link and it saves with a thumbnail.' : 'Write one line below. AI cleans it into a draft.'}
          style={{ marginTop: 32 }} />
      ) : null}

      {!loading && !empty && chip === 0 ? (
        <div style={{ display: 'flex', flexDirection: 'column', gap: 12 }}>
          {groups.map((g) => (
            <div key={g.name}>
              <div style={{ display: 'flex', alignItems: 'center', gap: 8, padding: '0 2px 7px' }}>
                <Avatar name={g.name} size={22} />
                <span style={{ flex: 1, font: '700 13px var(--font-ui)', color: 'var(--ink)' }}>{g.name}</span>
                <span style={{ flex: '0 0 auto', whiteSpace: 'nowrap', font: '600 12px var(--font-ui)', color: 'var(--slate-400)' }}>{g.items.length} {g.items.length === 1 ? 'post' : 'posts'}</span>
              </div>
              <Card pad={0}>
                {g.items.map((it, i) => (
                  <div key={it.id} style={{ display: 'flex', gap: 10, alignItems: 'center', padding: '9px 12px', borderBottom: i < g.items.length - 1 ? '1px solid var(--line)' : 'none' }}>
                    <Thumb format={it.format} w={34} h={45} radius={8} />
                    <div style={{ flex: 1, minWidth: 0 }}>
                      <div style={{ font: '700 13.5px var(--font-display)', letterSpacing: '-0.2px', color: 'var(--ink)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{it.title}</div>
                      <div style={{ marginTop: 2, font: '600 11.5px var(--font-ui)', color: 'var(--slate-400)' }}>{formatLabel(it.format)} · {it.when}</div>
                    </div>
                    <div style={{ textAlign: 'right' }}>
                      <div style={{ font: '700 14px var(--font-display)', letterSpacing: '-0.2px', color: 'var(--ink)' }}>{it.views}</div>
                      <div style={{ font: '600 10.5px var(--font-ui)', color: 'var(--slate-400)' }}>views</div>
                    </div>
                  </div>
                ))}
              </Card>
            </div>
          ))}
        </div>
      ) : null}

      {!loading && !empty && chip === 1 ? (
        <div style={{ display: 'flex', flexDirection: 'column', gap: 8 }}>
          {L.refs.map((it) => (
            <Card key={it.id} pad={12} style={{ display: 'flex', gap: 11, alignItems: 'center' }}>
              <Thumb format={it.format} w={40} h={52} />
              <div style={{ flex: 1, minWidth: 0 }}>
                <div style={{ font: '700 14px/1.3 var(--font-display)', letterSpacing: '-0.2px', color: 'var(--ink)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{it.title}</div>
                <div style={{ marginTop: 4, display: 'flex', gap: 6, alignItems: 'center' }}>
                  <FormatChip format={it.format} size="sm" />
                  <span style={{ font: '600 12px var(--font-ui)', color: 'var(--slate-400)' }}>@{it.handle} · {it.views}</span>
                </div>
              </div>
              <Icon name="link" size={16} color="var(--slate-300)" />
            </Card>
          ))}
        </div>
      ) : null}

      {!loading && !empty && chip === 2 ? (
        <div style={{ display: 'flex', flexDirection: 'column', gap: 8 }}>
          <div style={{ padding: 12, borderRadius: 'var(--radius-md)', background: 'var(--white)', border: `1.5px solid ${text ? 'var(--blue-500)' : 'var(--border-strong)'}`, boxShadow: text ? '0 0 0 3px rgba(27,166,238,0.18)' : 'none' }}>
            <textarea value={text} onChange={(e) => setText(e.target.value)} rows={text ? 2 : 1} placeholder="Type a post idea"
              style={{ width: '100%', boxSizing: 'border-box', border: 'none', outline: 'none', resize: 'none', background: 'transparent', font: '400 15px/1.4 var(--font-ui)', color: 'var(--ink)', padding: 0 }}></textarea>
            <div style={{ marginTop: 9, display: 'flex', alignItems: 'center', gap: 6 }}>
              {[['video', 'video', 'Video'], ['photo_carousel', 'images', 'Slideshow']].map(([k, icon, label]) => (
                <button key={k} type="button" onClick={() => setFmt(k)}
                  style={{ display: 'inline-flex', alignItems: 'center', gap: 5, padding: '7px 11px', borderRadius: 999, border: 'none', cursor: 'pointer', background: fmt === k ? 'var(--blue-500)' : 'var(--fill-quiet)', color: fmt === k ? 'var(--white)' : 'var(--slate-500)', font: '700 12px var(--font-ui)', transition: 'background var(--dur-fast) var(--ease-out), color var(--dur-fast) var(--ease-out), transform var(--dur-fast) var(--ease-out)', transform: fmt === k ? 'scale(1.04)' : 'scale(1)' }}>
                  <Icon name={icon} size={12} />{label}
                </button>
              ))}
              <span style={{ flex: 1 }}></span>
              <Button size="sm" variant="primary" disabled={!text.trim()} onClick={save}>Save</Button>
            </div>
          </div>
          {ideas.map((it) => (
            <Card key={it.id} pad={13} style={{ display: 'flex', gap: 11, alignItems: 'flex-start' }}>
              <Icon name={it.drafted ? 'sparkles' : 'zap'} size={15} color="var(--blue-600)" style={{ flex: '0 0 auto', marginTop: 2 }} />
              <div style={{ flex: 1, minWidth: 0 }}>
                <p style={{ margin: 0, font: '400 14px/1.45 var(--font-ui)', color: 'var(--ink)' }}>{it.body}</p>
                <div style={{ marginTop: 5, display: 'flex', gap: 7, alignItems: 'center', overflow: 'hidden', font: '600 12px var(--font-ui)', color: 'var(--slate-400)' }}>
                  {it.format ? <FormatChip format={it.format} size="sm" /> : null}
                  {it.drafted ? <span style={{ display: 'inline-flex', alignItems: 'center', gap: 4, whiteSpace: 'nowrap', color: 'var(--blue-700)', font: '700 11.5px var(--font-ui)' }}><Icon name="sparkles" size={11} />Cleaned with AI · draft ready</span> : null}
                  {!it.drafted ? <span style={{ whiteSpace: 'nowrap' }}>{it.age}</span> : null}
                </div>
              </div>
            </Card>
          ))}
        </div>
      ) : null}
    </AdminScreen>
  );
}

Object.assign(window, { LibraryScreen });
