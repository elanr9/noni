function longPress(fn, ms = 480) {
  let t = null;
  const start = () => { clearTimeout(t); t = setTimeout(fn, ms); };
  const stop = () => clearTimeout(t);
  return { onPointerDown: start, onPointerUp: stop, onPointerLeave: stop, onPointerCancel: stop, onContextMenu: (e) => { e.preventDefault(); stop(); fn(); } };
}

    if (!lib2IsLink(url)) url = 'https://www.tiktok.com/@queryqueen/video/73' + String(Date.now()).slice(-6);
    const ig = /instagram/i.test(url);
    const item = { id: 'r' + Date.now(), title: url.replace(/^https?:\/\/(www\.)?/, ''), handle: ig ? 'instagram' : 'tiktok', platform: ig ? 'Instagram' : 'TikTok', format: 'video', tone: 1, made: 0, resolving: true };
    setTimeout(() => {
      setPasting(false); setRefs((r) => [item, ...r]); setSub('unused'); flash('Reference saved');
      setTimeout(() => setRefs((r) => r.map((x) => x.id === item.id ? { ...x, resolving: false, title: 'Why your query gets a form rejection', handle: ig ? 'draft.two' : 'queryqueen' } : x)), 1600);
    }, 500);
  };
  const addMedia = (kind) => { setNameDraft(''); setSheet({ kind: 'name', mediaKind: kind }); };
  const saveMedia = () => { setMedia([{ id: 'm' + Date.now(), kind: sheet.mediaKind, dur: sheet.mediaKind === 'video' ? '0:12' : undefined, tone: 3, name: nameDraft.trim() }, ...media]); setSub(sheet.mediaKind); setSheet(null); flash(sheet.mediaKind === 'video' ? 'Recording added' : 'Screenshot added'); };

  const bump = (list, set, id) => set(list.map((x) => x.id === id ? { ...x, made: x.made + 1, madeOn: 'Today', post: `${LIB2.week.label} · slot ${LIB2.week.slot}` } : x));
  const make = (kind, item, type) => {
    setBusy(item.id);
    setTimeout(() => {
      setBusy(null); setBusyType(null); setSheet(null);
      if (kind === 'idea') bump(ideas, setIdeas, item.id);
      if (kind === 'ref') bump(refs, setRefs, item.id);
      if (kind === 'post') bump(ours, setOurs, item.id);
      onOpenPost && onOpenPost({ kind, item, type });
    }, 1400);
  };
  const pickType = (t) => { setBusyType(t.key); make(sheet.source.kind, sheet.source.item, t); };

  const lines = lib2Lines(text);
  const creators = [...new Set(LIB2.ours.map((o) => o.creator))];
  const oursShown = ours
    .filter((o) => ![...sel].some((k) => k.startsWith('c:')) || sel.has('c:' + o.creator))
    .filter((o) => ![...sel].some((k) => k.startsWith('t:')) || sel.has('t:' + o.type))
    .filter((o) => !q || o.title.toLowerCase().includes(q.toLowerCase()))
    .sort((a, b) => sort === 'Top 60d' ? b.viewsN - a.viewsN : a.id < b.id ? 1 : -1);
  const ideasSub = ideas.filter((i) => sub === 'used' ? i.made : !i.made);
  const ideasShown = ideasSub.filter((i) => !q || i.body.toLowerCase().includes(q.toLowerCase()));
  const refsSub = refs.filter((r) => sub === 'used' ? r.made : !r.made);
  const refsShown = refsSub.filter((r) => !q || r.title.toLowerCase().includes(q.toLowerCase()) || r.handle.includes(q.toLowerCase()));
  const mediaSub = media.filter((m) => m.kind === sub);
  const lanes = [['ideas', 'Ideas'], ['refs', 'References'], ['media', 'Media'], ['ours', 'Our posts']];

  const rowIdea = (it, i, last) => {
    const lp = longPress(() => setSheet({ kind: 'delete', what: 'idea', id: it.id }));
    const isEditing = editing === it.id;
    return (
      <div key={it.id} {...lp} style={{ display: 'flex', gap: 10, alignItems: 'flex-start', padding: '12px 14px', borderBottom: last ? 'none' : '1px solid var(--line)', animation: i === 0 ? 'noni-fade 240ms var(--ease-out) both' : 'none' }}>
        <div style={{ flex: 1, minWidth: 0 }}>
          {isEditing ? (
            <textarea autoFocus defaultValue={it.body} rows={2} onBlur={(e) => { const v = e.target.value.trim(); setIdeas(ideas.map((x) => x.id === it.id ? { ...x, body: v || x.body } : x)); setEditing(null); }}
              style={{ width: '100%', boxSizing: 'border-box', border: 'none', outline: 'none', resize: 'none', background: 'var(--blue-50)', borderRadius: 8, padding: '4px 6px', margin: '-4px -6px', font: '500 15px/1.4 var(--font-ui)', color: 'var(--ink)' }} />
          ) : <p onClick={() => setEditing(it.id)} style={{ margin: 0, font: '500 15px/1.4 var(--font-ui)', color: 'var(--ink)', cursor: 'text' }}>{it.body}</p>}
          {it.made ? <div style={{ marginTop: 5 }}><MadeMeta item={it} onOpenPost={() => onOpenPost && onOpenPost({ kind: 'idea', item: it })} /></div> : null}
          {killed && killed.id === it.id ? <KillNote reason={killed.reason} onClose={() => setKilled(null)} /> : null}
        </div>
        <MakeButton label={it.made ? 'Again' : 'Make'} busy={busy === it.id} onClick={() => setSheet({ kind: 'types', source: { kind: 'idea', item: it } })} />
      </div>
    );
  };
  const cardRef = (it) => {
    const lp = longPress(() => setSheet({ kind: 'delete', what: 'reference', id: it.id }));
    return (
      <Card key={it.id} pad={10} onClick={() => onOpenLink && onOpenLink(it)} {...lp} style={{ display: 'flex', gap: 11, alignItems: 'center' }}>
        {it.resolving ? <div style={{ width: 46, height: 62, borderRadius: 9, flex: '0 0 auto', background: 'var(--fill-quiet)', display: 'flex', alignItems: 'center', justifyContent: 'center' }}><Spinner size={14} color="var(--slate-300)" /></div> : <Thumb format={it.format} w={46} h={62} radius={9} cover={{ tone: it.tone, text: it.title }} />}
        <div style={{ flex: 1, minWidth: 0 }}>
          <div style={{ font: '700 14px/1.3 var(--font-display)', letterSpacing: '-0.2px', color: it.resolving ? 'var(--slate-400)' : 'var(--ink)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{it.resolving ? 'Fetching title' : it.title}</div>
          <div style={{ marginTop: 3, font: '600 12px var(--font-ui)', color: 'var(--slate-400)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>@{it.handle} · {it.platform}</div>
          {it.made ? <div style={{ marginTop: 5 }}><MadeMeta item={it} onOpenPost={() => onOpenPost && onOpenPost({ kind: 'ref', item: it })} /></div> : null}
          {killed && killed.id === it.id ? <KillNote reason={killed.reason} onClose={() => setKilled(null)} /> : null}
        </div>
        <MakeButton label={it.made ? 'Again' : 'Make'} busy={busy === it.id} onClick={() => setSheet({ kind: 'types', source: { kind: 'ref', item: it } })} />
      </Card>
    );
  };
  /* Our posts: one dense card, small thumbs, one line of meta. */
  const rowOurs = (it, i, last) => (
    <div key={it.id} onClick={() => onOpenLink && onOpenLink(it)} style={{ display: 'flex', gap: 10, alignItems: 'center', padding: '9px 12px', borderBottom: last ? 'none' : '1px solid var(--line)', cursor: 'pointer' }}>
      <Thumb format={it.format} w={34} h={45} radius={8} cover={{ tone: it.tone, text: it.title }} />
      <div style={{ flex: 1, minWidth: 0 }}>
        <div style={{ font: '700 13.5px var(--font-display)', letterSpacing: '-0.2px', color: 'var(--ink)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{it.title}</div>
        <div style={{ marginTop: 2, display: 'flex', alignItems: 'center', gap: 5, font: '600 11.5px var(--font-ui)', color: 'var(--slate-400)', whiteSpace: 'nowrap', overflow: 'hidden' }}>
          <span style={{ color: 'var(--ink)' }}>{it.creator}</span>·<span>{LIB2.types.find((t) => t.key === it.type).label}</span>·<span>{window.formatLabel(it.format)}</span>·<span>{it.when}</span>
        </div>
        {killed && killed.id === it.id ? <KillNote reason={killed.reason} onClose={() => setKilled(null)} /> : null}
      </div>
      <div style={{ textAlign: 'right', flex: '0 0 auto' }}><div style={{ font: '700 13.5px var(--font-display)', letterSpacing: '-0.2px', color: 'var(--green)' }}>{it.views}</div><div style={{ font: '600 10px var(--font-ui)', color: 'var(--slate-400)' }}>views</div></div>
      <MakeButton label="Remake" busy={busy === it.id} onClick={() => make('post', it, LIB2.types.find((t) => t.key === it.type))} />
    </div>
  );
  /* Media: two columns, title under every tile. */
  const tileMedia = (it) => {
    const lp = longPress(() => setSheet({ kind: 'delete', what: 'media', id: it.id }));
    return (
      <button key={it.id} type="button" {...lp} onClick={() => setSheet({ kind: 'preview', item: it })} style={{ border: 'none', padding: 0, background: 'transparent', cursor: 'pointer', textAlign: 'left', minWidth: 0 }}>
        <div style={{ position: 'relative', aspectRatio: '4 / 5', borderRadius: 12, overflow: 'hidden', background: lib2Cover(it.tone), boxShadow: 'var(--shadow-media)' }}>
          {it.kind === 'video' ? <>
            <span style={{ position: 'absolute', inset: 0, display: 'flex', alignItems: 'center', justifyContent: 'center' }}><span style={{ width: 36, height: 36, borderRadius: 999, background: 'rgba(255,255,255,0.88)', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}><Icon name="play" size={16} color="var(--ink)" /></span></span>
            <span style={{ position: 'absolute', right: 8, bottom: 8, padding: '2px 7px', borderRadius: 999, background: 'rgba(11,15,20,0.6)', color: 'var(--white)', font: '700 10.5px var(--font-ui)' }}>{it.dur}</span>
          </> : null}
        </div>
        <div style={{ marginTop: 7, display: 'flex', alignItems: 'center', gap: 5, padding: '0 2px' }}>
          <Icon name={it.kind === 'video' ? 'video' : 'images'} size={12} color="var(--slate-400)" />
          <span style={{ flex: 1, minWidth: 0, font: '700 13px var(--font-ui)', color: 'var(--ink)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{it.name}</span>
        </div>
      </button>
    );
  };

  const filterCount = sel.size;
  const laneEmpty = lane === 'ideas' ? !ideas.length : lane === 'refs' ? !refs.length : lane === 'ours' ? !ours.length : !media.length;
  return (
    <div style={{ flex: 1, minHeight: 0, position: 'relative', display: 'flex', flexDirection: 'column', background: 'var(--off-white)' }}>
      {/* Pinned: title, lanes, the lane's entry point, its toolbar */}
      <div style={{ flex: '0 0 auto', padding: `6px ${GUT}px 10px`, display: 'flex', flexDirection: 'column', gap: 10, background: 'var(--off-white)', boxShadow: '0 8px 12px -10px rgba(15,23,32,0.08)', zIndex: 2 }}>
        <div style={{ display: 'flex', alignItems: 'center', gap: 10 }}>
          <h1 style={{ margin: 0, flex: 1, font: '700 30px var(--font-display)', letterSpacing: 'var(--tracking-title)', color: 'var(--ink)' }}>Library</h1>
          {refreshing ? <span style={{ display: 'inline-flex', alignItems: 'center', gap: 6, font: '600 12px var(--font-ui)', color: 'var(--slate-400)' }}><Spinner size={13} color="var(--slate-400)" />Refreshing</span> : null}
        </div>
        <div style={{ display: 'flex', gap: 3, padding: 3, borderRadius: 999, background: 'var(--fill-quiet)' }}>
          {lanes.map(([k, label]) => {
            const on = lane === k;
            return <button key={k} type="button" onClick={() => switchLane(k)} aria-pressed={on} style={{ flex: 1, height: 36, padding: '0 4px', border: 'none', cursor: 'pointer', borderRadius: 999, background: on ? 'var(--white)' : 'transparent', boxShadow: on ? 'var(--shadow-card)' : 'none', color: on ? 'var(--ink)' : 'var(--slate-500)', font: '700 13px var(--font-ui)', letterSpacing: '-0.1px', whiteSpace: 'nowrap', transition: 'background var(--dur-fast) var(--ease-out), color var(--dur-fast) var(--ease-out)' }}>{label}</button>;
          })}
        </div>

        {lane === 'ideas' ? (
          <div style={{ padding: '10px 10px 10px 14px', borderRadius: 'var(--radius-md)', background: 'var(--white)', border: `1.5px solid ${text ? 'var(--blue-500)' : 'var(--border-strong)'}`, boxShadow: text ? '0 0 0 3px rgba(27,166,238,0.18)' : 'var(--shadow-card)', transition: 'box-shadow var(--dur-fast) var(--ease-out), border-color var(--dur-fast) var(--ease-out)' }}>
            <div style={{ display: 'flex', gap: 8, alignItems: 'flex-end' }}>
              <textarea value={text} onChange={(e) => setText(e.target.value)} rows={text ? Math.min(4, Math.max(1, lines.length)) : 1} placeholder="Type an idea"
                style={{ flex: 1, minWidth: 0, border: 'none', outline: 'none', resize: 'none', background: 'transparent', font: '500 15px/1.4 var(--font-ui)', color: 'var(--ink)', padding: '4px 0' }} />
              <button type="button" disabled={!lines.length} onClick={saveIdeas} aria-label="Save" style={{ width: 36, height: 36, flex: '0 0 auto', borderRadius: 999, border: 'none', cursor: lines.length ? 'pointer' : 'default', background: lines.length ? 'var(--blue-500)' : 'var(--fill-quiet)', color: lines.length ? 'var(--white)' : 'var(--slate-300)', display: 'inline-flex', alignItems: 'center', justifyContent: 'center', transition: 'background var(--dur-fast) var(--ease-out), transform var(--dur-fast) var(--ease-out)', transform: lines.length ? 'scale(1.04)' : 'scale(1)' }}><Icon name="arrow-right" size={18} /></button>
            </div>
            {lines.length > 1 ? <div style={{ marginTop: 6, font: '700 12px var(--font-ui)', color: 'var(--blue-700)', animation: 'noni-fade 160ms var(--ease-out) both' }}>Saves as {lines.length} ideas, one per line</div> : null}
          </div>
        ) : null}

        {lane === 'refs' ? (
          <div style={{ display: 'flex', alignItems: 'center', gap: 10, padding: '8px 8px 8px 14px', borderRadius: 'var(--radius-md)', background: 'var(--white)', border: '1.5px solid var(--border-strong)', boxShadow: 'var(--shadow-card)' }}>
            <Icon name="link" size={17} color="var(--slate-400)" />
            <div style={{ flex: 1, minWidth: 0 }}>
              <div style={{ font: '600 15px var(--font-ui)', color: 'var(--ink)' }}>Paste a link</div>
              <div style={{ font: '600 12px var(--font-ui)', color: 'var(--slate-400)' }}>TikTok or Instagram, copied from the app</div>
            </div>
            <Button size="sm" variant="primary" icon={pasting ? undefined : 'copy'} disabled={pasting} onClick={pasteLink} style={{ minWidth: 92 }}>{pasting ? 'Pasting' : 'Paste'}</Button>
          </div>
        ) : null}

        {lane === 'ideas' || lane === 'refs' ? <SubTabs active={sub} onSelect={setSub} items={lane === 'ideas' ? [['unused', 'Unused', ideas.filter((i) => !i.made).length], ['used', 'Used', ideas.filter((i) => i.made).length]] : [['unused', 'Unused', refs.filter((r) => !r.made).length], ['used', 'Used', refs.filter((r) => r.made).length]]} /> : null}
        {lane === 'media' ? <SubTabs active={sub} onSelect={setSub} items={[['image', 'Screenshots', media.filter((m) => m.kind === 'image').length], ['video', 'Recordings', media.filter((m) => m.kind === 'video').length]]} /> : null}

        {!loading && !laneEmpty ? (
          <div style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
            {lane === 'ideas' ? <LibSearch value={q} onChange={setQ} placeholder={sub === 'used' ? 'Search used ideas' : 'Search unused ideas'} /> : null}
            {lane === 'refs' ? <LibSearch value={q} onChange={setQ} placeholder={sub === 'used' ? 'Search used references' : 'Search unused references'} /> : null}
            {lane === 'ours' ? <>
              <LibSearch value={q} onChange={setQ} placeholder="Search posts" />
              <button type="button" onClick={() => setSort(sort === 'Top 60d' ? 'Recent' : 'Top 60d')} style={{ height: 38, padding: '0 11px', borderRadius: 999, border: 'none', cursor: 'pointer', background: 'transparent', color: 'var(--slate-500)', font: '700 13px var(--font-ui)', display: 'inline-flex', alignItems: 'center', gap: 4, whiteSpace: 'nowrap' }}>{sort}<Icon name="chevrons-up-down" size={13} /></button>
              <button type="button" onClick={() => setSheet({ kind: 'filter' })} aria-label="Filter" style={{ position: 'relative', width: 38, height: 38, flex: '0 0 auto', borderRadius: 999, border: 'none', cursor: 'pointer', background: filterCount ? 'var(--blue-100)' : 'var(--fill-quiet)', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}>
                <Icon name="layout-list" size={16} color={filterCount ? 'var(--blue-700)' : 'var(--slate-500)'} />
                {filterCount ? <span style={{ position: 'absolute', top: -3, right: -3, minWidth: 17, height: 17, padding: '0 4px', borderRadius: 999, background: 'var(--blue-500)', color: 'var(--white)', font: '700 10.5px var(--font-ui)', display: 'inline-flex', alignItems: 'center', justifyContent: 'center', boxShadow: '0 0 0 2px var(--off-white)' }}>{filterCount}</span> : null}
              </button>
            </> : null}
            {lane === 'media' ? <><span style={{ flex: 1, font: '600 13px var(--font-ui)', color: 'var(--slate-500)', padding: '0 2px' }}>{mediaSub.length} {sub === 'video' ? (mediaSub.length === 1 ? 'recording' : 'recordings') : (mediaSub.length === 1 ? 'screenshot' : 'screenshots')}</span><Button size="sm" variant="primary" icon="plus" onClick={() => addMedia(sub)}>{sub === 'video' ? 'Add recording' : 'Add screenshot'}</Button></> : null}
          </div>
        ) : null}
      </div>

      {/* Scrolling lane */}
      <div style={{ flex: 1, minHeight: 0, overflowY: 'auto', padding: `4px ${GUT}px 118px` }}>
        {loading ? <LaneSkeleton lane={lane} /> : null}
        {!loading && lane === 'ideas' ? (ideas.length ? (ideasShown.length ? <Card pad={0} style={{ overflow: 'hidden' }}>{ideasShown.map((it, i) => rowIdea(it, i, i === ideasShown.length - 1))}</Card> : (ideasSub.length ? <p style={{ margin: '32px 0', textAlign: 'center', font: '500 14px var(--font-ui)', color: 'var(--slate-400)' }}>No ideas match</p> : <LaneEmpty lane="ideas" sub={sub} />)) : <LaneEmpty lane="ideas" sub={sub} />) : null}
        {!loading && lane === 'refs' ? (refs.length ? (refsShown.length ? <div style={{ display: 'flex', flexDirection: 'column', gap: 8 }}>{refsShown.map(cardRef)}</div> : (refsSub.length ? <p style={{ margin: '32px 0', textAlign: 'center', font: '500 14px var(--font-ui)', color: 'var(--slate-400)' }}>No references match</p> : <LaneEmpty lane="refs" sub={sub} />)) : <LaneEmpty lane="refs" sub={sub} />) : null}
        {!loading && lane === 'ours' ? (ours.length ? (oursShown.length ? <Card pad={0} style={{ overflow: 'hidden' }}>{oursShown.map((it, i) => rowOurs(it, i, i === oursShown.length - 1))}</Card> : <p style={{ margin: '32px 0', textAlign: 'center', font: '500 14px var(--font-ui)', color: 'var(--slate-400)' }}>No posts match these filters</p>) : <LaneEmpty lane="ours" />) : null}
        {!loading && lane === 'media' ? (mediaSub.length ? (
          <div style={{ display: 'grid', gridTemplateColumns: 'repeat(2,1fr)', gap: 12 }}>
            {mediaSub.map(tileMedia)}
            <button type="button" onClick={() => addMedia(sub)} style={{ border: 'none', padding: 0, background: 'transparent', cursor: 'pointer', textAlign: 'left' }}>
              <div style={{ aspectRatio: '4 / 5', borderRadius: 12, border: '1.5px dashed var(--border-strong)', display: 'flex', flexDirection: 'column', alignItems: 'center', justifyContent: 'center', gap: 8, color: 'var(--slate-500)', font: '700 13px var(--font-ui)' }}><Icon name={sub === 'video' ? 'video' : 'image-plus'} size={24} color="var(--blue-500)" />{sub === 'video' ? 'Add recording' : 'Add screenshot'}</div>
            </button>
          </div>
        ) : <LaneEmpty lane="media" sub={sub} onAdd={() => addMedia(sub)} />) : null}
        {!loading && more && lane !== 'media' ? <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'center', gap: 8, padding: '18px 0 6px', font: '600 13px var(--font-ui)', color: 'var(--slate-400)' }}><Spinner size={14} color="var(--slate-300)" />Loading more</div> : null}
      </div>

      {toast ? <Toast text={toast} /> : null}
      {sheet && sheet.kind === 'types' ? <TypeSheet source={sheet.source} busyKey={busyType} onPick={pickType} onClose={() => !busyType && setSheet(null)} animate={animate} /> : null}
      {sheet && sheet.kind === 'filter' ? <FilterSheet creators={creators} types={LIB2.types} sel={sel} onChange={setSel} count={oursShown.length} onClose={() => setSheet(null)} animate={animate} /> : null}
      {sheet && sheet.kind === 'preview' ? <MediaPreview item={sheet.item} onClose={() => setSheet(null)} onRename={(n) => setMedia(media.map((m) => m.id === sheet.item.id ? { ...m, name: n } : m))} /> : null}
      {sheet && sheet.kind === 'name' ? <NameSheet kind={sheet.mediaKind} value={nameDraft} onChange={setNameDraft} onSave={saveMedia} onClose={() => setSheet(null)} animate={animate} /> : null}
      {sheet && sheet.kind === 'delete' ? <DeleteSheet what={sheet.what} animate={animate} note={sheet.what === 'media' ? 'Posts already using it keep their copy.' : 'Posts made from it are not affected.'} onClose={() => setSheet(null)}
        onDelete={() => { if (sheet.what === 'idea') setIdeas(ideas.filter((x) => x.id !== sheet.id)); if (sheet.what === 'reference') setRefs(refs.filter((x) => x.id !== sheet.id)); if (sheet.what === 'media') setMedia(media.filter((x) => x.id !== sheet.id)); setSheet(null); }} /> : null}
    </div>
  );
}
