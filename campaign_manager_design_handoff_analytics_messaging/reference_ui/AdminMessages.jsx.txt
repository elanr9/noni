/* Left-aligned message row. Slack rhythm: avatar, name and time on one line,
   body under it. Consecutive messages from one sender collapse the header. */
function MsgRow({ m, prev, children }) {
  const { Icon } = window.NoniDesignSystem_710e43;
  const mine = m.from === 'me';
  const name = mine ? 'You' : m.from.name;
  const same = prev && !prev.day && prev.from === m.from;
  return (
    <div style={{ display: 'flex', gap: 10, padding: same ? '2px 0 0 0' : '10px 0 0 0' }}>
      <span style={{ width: 34, flex: '0 0 auto' }}>{!same ? <Avatar name={mine ? 'You' : m.from.name} size={34} tone={mine || m.from.role ? 'quiet' : 'brand'} /> : null}</span>
      <div style={{ flex: 1, minWidth: 0 }}>
        {!same ? <div style={{ display: 'flex', alignItems: 'baseline', gap: 7 }}><span style={{ font: '700 14px var(--font-display)', letterSpacing: '-0.2px', color: 'var(--ink)' }}>{name}</span><span style={{ font: '600 11px var(--font-ui)', color: 'var(--slate-400)' }}>{m.time}</span></div> : null}
        {children}
      </div>
    </div>
  );
}

/* A creator's thread. Slack rows, posts as cards, day dividers. The header
   carries the creator's live state so you never leave to check it. */

  const { Icon } = window.NoniDesignSystem_710e43;
  const D = MSG_DATA();
  const isMember = !!creator.role;
  const msgs = isMember ? [{ day: 'Today' }, { from: creator, time: '08:44', body: D.dms.find((d) => d.who === creator).last }, { from: 'me', time: '08:51', body: 'On it. Two posts and a song check left in the queue.' }] : D.thread;
  const [attach, setAttach] = React.useState(false);
  return (
    <div style={{ flex: 1, minHeight: 0, display: 'flex', flexDirection: 'column', background: 'var(--white)' }}>
      <div style={{ flex: '0 0 auto', paddingTop: 4, borderBottom: '1px solid var(--line)', background: 'var(--white)' }}>
        <PushHeader title={creator.name} meta={isMember ? creator.role : `@${creator.tiktok} · 2 posts in review · 3 live this week`} onBack={onBack} style={{ background: 'transparent' }}
          right={!isMember ? <button type="button" onClick={onProfile} aria-label="Profile" style={{ border: 'none', background: 'transparent', padding: 0, cursor: 'pointer', display: 'inline-flex' }}><Avatar name={creator.name} size={34} /></button> : null} />
      </div>
      <div style={{ flex: 1, minHeight: 0, overflowY: 'auto', padding: `4px ${window.GUT}px 16px` }}>
        {msgs.map((m, i) => {
          if (m.day) return <DayDivider key={i} label={m.day} />;
          return (
            <MsgRow key={i} m={m} prev={msgs[i - 1]}>
              {m.body ? <p style={{ margin: '2px 0 0', font: '400 14.5px/1.45 var(--font-ui)', color: 'var(--ink)' }}>{m.body}</p> : null}
              {m.post ? <PostCard post={m.post} event={m.event} label={m.label} review={m.review} onReview={() => onReview && onReview(m.post)} onOpen={() => onOpenPost && onOpenPost(m.post)} /> : null}
              {m.media ? <div style={{ marginTop: 6, width: 168, height: 118, borderRadius: 12, background: 'linear-gradient(160deg,#E7F4FD 0%,#DCE7F0 100%)', display: 'flex', alignItems: 'center', justifyContent: 'center' }}><Icon name="images" size={22} color="var(--blue-300)" /></div> : null}
            </MsgRow>
          );
        })}
      </div>
      {attach ? (
        <div style={{ display: 'flex', gap: 8, padding: `10px ${window.GUT}px 0`, borderTop: '1px solid var(--line)', background: 'var(--white)' }}>
          {[['layout-list', 'A post'], ['images', 'Photo'], ['camera', 'Camera']].map(([icon, label]) => <button key={label} type="button" onClick={() => setAttach(false)} style={{ flex: 1, display: 'flex', flexDirection: 'column', alignItems: 'center', gap: 6, padding: '11px 0', borderRadius: 'var(--radius-md)', border: 'none', cursor: 'pointer', background: 'var(--blue-50)', font: '700 12px var(--font-ui)', color: 'var(--blue-700)' }}><Icon name={icon} size={18} color="var(--blue-600)" />{label}</button>)}
        </div>
      ) : null}
      <Composer placeholder={`Message ${creator.short || creator.name.split(' ')[0]}`} onAttach={() => setAttach(!attach)} />
    </div>
  );
}
