const CLIP_SRC = (k) => `roll${(k * 5 + 3) % 15}`;

/* Clip plan per kind. Handheld: one clip per point, plug randomized. Story:
   hook clip, then free clips, then the plug. Freestyle: free clips, then the plug. */
function clipPlan(post) {
  const B = window.SS.reel;
  if (post.type === 'Long text') return [{ id: 0, role: 'cover', text: window.SS.longtext, say: 'Six seconds. Look at the camera, the text does the talking.', target: 6, tstyle: { size: 0.6, align: 'left', bg: 'none' }, box: { x: 0.5, y: 0.42, w: 0.86 }, clip: null }];
  if (post.kind === 'handheld') return creatorSlides(B.slides).map((s) => ({ ...s, say: s.shot, clip: null }));
  if (post.kind === 'story') return [{ id: 0, role: 'cover', text: post.title, say: B.hookSay, clip: null }];
  return [];
}
