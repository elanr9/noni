/* Four tiny slides. Text box = words you wrote. Small card = your screenshot.
   Dashed camera slot = the creator's photo. */
function MiniDeck({ kind }) {
  const { Icon } = NS();
  const n = 4; const w = 30; const h = 53;
  const TextBox = ({ lines = 2 }) => <span style={{ display: 'flex', flexDirection: 'column', gap: 2, padding: '3px 4px', borderRadius: 3, background: '#fff', boxShadow: '0 1px 3px rgba(0,0,0,0.22)' }}>{Array.from({ length: lines }).map((_, i) => <span key={i} style={{ height: 2.5, borderRadius: 2, background: 'var(--ink)', width: i === lines - 1 ? '65%' : '100%' }} />)}</span>;
  const ShotBox = () => <span style={{ display: 'block', width: '70%', margin: '0 auto', borderRadius: 3, overflow: 'hidden', boxShadow: '0 1px 3px rgba(0,0,0,0.22)' }}><span style={{ display: 'block', height: 4, background: '#2F6FD1' }} /><span style={{ display: 'block', height: 12, background: '#DCE9F7' }} /></span>;
  return (
    <span style={{ display: 'flex', gap: 4, flex: '0 0 auto' }}>
      {Array.from({ length: n }).map((_, i) => {
        const cta = kind !== 'handheld' && i === n - 1;
        const text = kind === 'handheld' || (kind === 'story' && i === 0) || cta;
        const shot = (kind === 'handheld' && i % 2 === 1) || cta;
        return (
          <span key={i} style={{ position: 'relative', width: w, height: h, borderRadius: 7, overflow: 'hidden', background: cta ? 'linear-gradient(180deg,#CFDCE8,#4F6276)' : 'repeating-linear-gradient(135deg,#F3F7FB 0 4px,#EAF0F6 4px 8px)', border: cta ? 'none' : '1.5px dashed var(--line-strong)', display: 'flex', flexDirection: 'column', justifyContent: 'center', gap: 4, padding: 4 }}>
            {!text && !shot ? <span style={{ display: 'flex', justifyContent: 'center' }}><Icon name="camera" size={14} color="var(--slate-300)" /></span> : null}
            {shot ? <ShotBox /> : null}
            {text ? <TextBox lines={shot ? 1 : 2} /> : null}
          </span>
        );
      })}
    </span>
  );
}

function HandheldEditor({ slot, onBack, onSave, initial, onDark, kind = 'handheld', postType, format = 'slideshow' }) {

  const reel = format === 'reel';
  const lt = reel && postType && postType.id === 'longtext';
  const D = reel ? window.SS.reel : window.SS.handheld;
  const story = kind === 'story';
  const tipsStory = story && postType && postType.id === 'tips';
  const [count, setCountN] = React.useState(null);
  const seed = (arr) => { const a = arr.map((s, i) => ({ ...s, id: i })); if (story) return a; const c = a.findIndex((s) => s.role === 'cta'); if (c > 0) { const [cta] = a.splice(c, 1); a.splice(3, 0, cta); } return a; };
  const STEPS = story ? [{ id: 'cover', name: reel ? 'Hook' : 'Cover' }, { id: 'cta', name: 'CTA' }, { id: 'details', name: 'Details' }, { id: 'review', name: 'Review' }] : lt ? [{ id: 'cover', name: 'Text' }, { id: 'cta', name: 'CTA line' }, { id: 'details', name: 'Details' }, { id: 'review', name: 'Review' }] : [{ id: 'cover', name: reel ? 'Hook' : 'Cover' }, { id: 'points', name: reel ? 'Points' : 'Slides' }, { id: 'details', name: 'Details' }, { id: 'review', name: 'Review' }];
  const [step, setStep] = React.useState(initial ? 'review' : 'cover');
  const stepIdx = STEPS.findIndex((s) => s.id === step);
  const [gap, setGap] = React.useState({ min: 4, max: 8 });
  const [gapOpen, setGapOpen] = React.useState(false);
  const [slides, setSlides] = React.useState(initial ? seed(D.slides) : [{ id: 0, role: 'cover', text: lt && initial ? window.SS.longtext : '', shot: '', inset: null, tstyle: lt ? { size: 0.62, align: 'left', bg: 'none' } : undefined, box: lt ? { x: 0.5, y: 0.42, w: 0.86 } : undefined }, { id: 1, role: 'cta', text: '', shot: '', inset: null }]);
  const ctaPos = React.useRef(2 + Math.floor(Math.random() * 4));
  const INTRO_KEY = 'noni.ss.ctaIntroSeen.manager';
  const introSeen = React.useRef(Boolean(localStorage.getItem(INTRO_KEY)));
  const showIntro = () => { if (introSeen.current) return false; introSeen.current = true; localStorage.setItem(INTRO_KEY, '1'); setSheet('intro'); return true; };
  const [sel, setSel] = React.useState(0);
  const [search, setSearch] = React.useState('');
  const [caption, setCaption] = React.useState('');
  const [tags, setTags] = React.useState('');
  const [sheet, setSheet] = React.useState(null);
  const [busy, setBusy] = React.useState(null);
  React.useEffect(() => { onDark && onDark(step !== 'details' || sheet === 'fyp'); }, [step, sheet]);
  const [reword, setReword] = React.useState(true);
  const nextId = React.useRef(100);
  const ctaIdx = slides.findIndex((s) => s.role === 'cta');
  const cur = slides[sel];
  const title = slides[0]?.text || '';
  const points = slides.filter((s) => s.role === 'point');
  const ctaOk = Boolean(slides[ctaIdx]?.text?.trim());
  const pointsOk = points.length > 0 && points.every((p) => p.text.trim()) && (story || ctaOk);
  const lastIdx = slides.length - 1;
  const onLastPoint = step === 'points' && sel >= lastIdx;
  const onCta = !story && step === 'points' && cur?.role === 'cta';
  React.useEffect(() => { if (onCta && !ctaOk) showIntro(); }, [onCta]);
  const patchAt = (at, p) => setSlides((arr) => arr.map((s, i) => i === at ? { ...s, ...p } : s));
  const setCount = (n) => { setSlides((arr) => { const cov = arr[0], cta = arr.find((s) => s.role === 'cta'), m = cta && !story ? Math.max(1, n - 1) : n, pts = arr.filter((s) => s.role === 'point').slice(0, m); while (pts.length < m) pts.push({ id: nextId.current++, role: 'point', text: '', shot: '', inset: null }); if (!cta) return [cov, ...pts]; if (story) return [cov, ...pts, cta]; const at = Math.min(ctaPos.current, pts.length + 1) - 1; pts.splice(at, 0, cta); return [cov, ...pts]; }); setSel(1); };
  const go = (id) => { setStep(id); if (id === 'cover') setSel(0); if (id === 'cta') { setSel(ctaIdx); if (tipsStory && !count) setSheet('count'); else if (!ctaOk) showIntro(); } if (id === 'points') { if (points.length) setSel(1); else setSheet('count'); } };
  const isCta = (s) => s.role === 'cta';
  const next = () => { if (step === 'points' && !onLastPoint) { setSel(sel + 1); return; } go(STEPS[stepIdx + 1].id); };
  const generateCta = () => { const it = matchFeature(title, postType, story || lt ? null : ctaIdx); setBusy('cta'); setTimeout(() => { patchAt(ctaIdx, lt ? { text: it.cta, inset: null, genAt: Date.now() } : { text: it.cta, inset: it.id, box: { x: 0.44, y: 0.56, w: 0.66 }, insetBox: { x: 0.44, y: 0.27, w: 0.42 }, boxIg: { x: 0.5, y: 0.64, w: 0.72 }, insetBoxIg: { x: 0.5, y: 0.38, w: 0.42 }, genAt: Date.now() }); setBusy(null); }, 1700); };

/* Hashtags as chips. Space, comma or enter adds; backspace on empty removes the last. */
function TagField({ tags, onChange }) {
  const { Icon } = NS();
  const [v, setV] = React.useState('');
  const ref = React.useRef(null);
  const add = (raw) => { const t = raw.replace(/^#+/, '').replace(/[^\w]/g, '').toLowerCase(); if (t && !tags.includes(t)) onChange([...tags, t]); setV(''); };
  const onKey = (e) => { if (e.key === ' ' || e.key === ',' || e.key === 'Enter') { e.preventDefault(); add(v); } else if (e.key === 'Backspace' && !v && tags.length) onChange(tags.slice(0, -1)); };
  return (
    <div onClick={() => ref.current && ref.current.focus()} style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'center', gap: 8, minHeight: 56, padding: '10px 14px', borderRadius: 'var(--radius-lg)', background: 'var(--white)', border: '1px solid var(--border)', cursor: 'text' }}>
      {tags.map((t) => <span key={t} style={{ display: 'inline-flex', alignItems: 'center', gap: 4, height: 32, padding: '0 6px 0 11px', borderRadius: 999, background: 'var(--blue-100)', color: 'var(--blue-700)', font: '700 14px var(--font-ui)' }}>#{t}<button type="button" onClick={(e) => { e.stopPropagation(); onChange(tags.filter((x) => x !== t)); }} aria-label={`Remove ${t}`} style={{ width: 22, height: 22, borderRadius: 999, border: 'none', background: 'transparent', cursor: 'pointer', display: 'inline-flex', alignItems: 'center', justifyContent: 'center', padding: 0 }}><Icon name="x" size={13} color="var(--blue-700)" /></button></span>)}
      <input ref={ref} value={v} onChange={(e) => setV(e.target.value)} onKeyDown={onKey} onBlur={() => v && add(v)} placeholder={tags.length ? '' : 'collegesoccer'} style={{ flex: 1, minWidth: 90, border: 'none', outline: 'none', background: 'transparent', font: '500 16px var(--font-ui)', color: 'var(--ink)', padding: '4px 0', height: 32 }} />
    </div>
  );
}

/* ---------- Shell ---------- */
function SlideshowBriefs({ start = 'week', label, format = 'slideshow' }) {
  window.SS.activeFormat = format;
  const reel = format === 'reel';
  const PT = window.SS.postTypes[reel ? 'reel' : 'handheld'];
  const [slots, setSlots] = React.useState(reel ? window.SS.week.reelSlots : window.SS.week.slots);
  const [screen, setScreen] = React.useState(start === 'week' ? { name: 'week' } : start === 'kind' ? { name: 'kind', slot: reel ? 2 : 1 } : { name: 'edit', kind: start, slot: reel ? 2 : 1, initial: start === 'handheld', postType: start === 'handheld' ? PT[1] : start === 'story' ? PT[0] : null });
  const [toast, setToast] = React.useState(null);
  const [dark, setDark] = React.useState(false);
  React.useEffect(() => { if (!(screen.name === 'edit' && (screen.kind === 'handheld' || screen.kind === 'story'))) setDark(false); }, [screen]);
  const save = (post) => { setSlots((arr) => arr.map((s, i) => i === screen.slot ? post : s)); setScreen({ name: 'week' }); setToast(`Added to slot ${String(screen.slot + 1).padStart(2, '0')}`); setTimeout(() => setToast(null), 2200); };
  const back = () => setScreen(screen.name === 'edit' ? { name: 'kind', slot: screen.slot } : { name: 'week' });
  let body;
  if (screen.name === 'week') body = <WeekScreen format={format} slots={slots} onNew={(i) => setScreen({ name: 'kind', slot: i })} onOpen={(i) => setScreen({ name: 'edit', kind: slots[i].kind, slot: i, initial: true, postType: PT.find((t) => t.name === slots[i].type) })} />;
  else if (screen.name === 'kind') body = <KindScreen format={format} slot={screen.slot} onBack={back} onPick={(k, t) => k === 'open' ? save({ title: 'Freestyle', kind: 'open', count: '' }) : setScreen({ name: 'edit', kind: k, slot: screen.slot, postType: t })} />;
  else if (screen.kind === 'handheld' || screen.kind === 'story') body = <HandheldEditor key={screen.kind} format={format} slot={screen.slot} onBack={back} onSave={save} initial={screen.initial && screen.kind === 'handheld'} onDark={setDark} kind={screen.kind} postType={screen.postType || (screen.kind === 'handheld' ? PT[1] : PT[0])} />;
  else body = <OpenEditor slot={screen.slot} onBack={back} onSave={save} />;
  return (
    <Phone label={label} dark={dark}>
      <div key={`${screen.name}-${screen.kind}-${screen.slot}`} style={{ flex: 1, minHeight: 0, display: 'flex', flexDirection: 'column', animation: 'noni-screen-in var(--dur-base) var(--ease-out) both', position: 'relative' }}>{body}</div>
      {toast ? <div style={{ position: 'absolute', left: 20, right: 20, bottom: 40, padding: '12px 16px', borderRadius: 16, background: 'var(--ink)', color: '#fff', font: '600 14px var(--font-ui)', boxShadow: 'var(--shadow-float)', zIndex: 90, display: 'flex', alignItems: 'center', gap: 10, animation: 'noni-toast-in var(--dur-base) var(--ease-out) both' }}><window.NoniDesignSystem_710e43.Icon name="check" size={16} color="#fff" />{toast}</div> : null}
    </Phone>
  );
}
