function useDrag(W, H, onMove, onEnd) {
  const st = React.useRef(null);
  return {
    onPointerDown: (e) => { st.current = { x: e.clientX, y: e.clientY, moved: false }; e.currentTarget.setPointerCapture(e.pointerId); e.stopPropagation(); },
    onPointerMove: (e) => { if (!st.current) return; const dx = (e.clientX - st.current.x) / W, dy = (e.clientY - st.current.y) / H; if (Math.abs(dx) + Math.abs(dy) > 0.002) st.current.moved = true; st.current = { ...st.current, x: e.clientX, y: e.clientY }; onMove(dx, dy); },
    onPointerUp: (e) => { const m = st.current?.moved; st.current = null; onEnd && onEnd(m, e); },
    onPointerCancel: () => { st.current = null; },
  };
}

function textBoxStyle(ts, fs) {
  const f = FONTS[ts.font] || FONTS.modern;

}

/* Per-range styling inside one box: spans = [{ start, end, color?, font? }]. */
function runsOf(text, spans, ts, fs) {
  const t = text || '';
  if (!spans || !spans.length) return [<span key="a">{t}</span>];
  const per = new Array(t.length).fill(null);
  spans.forEach((sp) => { for (let i = Math.max(0, sp.start); i < Math.min(t.length, sp.end); i++) per[i] = { ...(per[i] || {}), ...sp }; });
  const out = []; let i = 0;
  while (i < t.length) { let j = i + 1; const k = JSON.stringify(per[i]); while (j < t.length && JSON.stringify(per[j]) === k) j++; const sp = per[i]; const st = {}; if (sp) { if (sp.color) { st.color = sp.color; if (ts.bg === 'outline') st.WebkitTextStroke = `${Math.max(1, fs * 0.055)}px ${light(sp.color) ? '#000' : '#fff'}`; } if (sp.font && FONTS[sp.font]) { st.font = FONTS[sp.font].font; st.fontSize = fs; st.letterSpacing = FONTS[sp.font].ls; } } out.push(<span key={i} style={st}>{t.slice(i, j)}</span>); i = j; }
  return out;
}
function shiftSpans(spans, pos, delta) {
  if (!spans || !delta) return spans;
  return spans.map((sp) => { if (sp.start >= pos) return { ...sp, start: Math.max(pos, sp.start + delta), end: Math.max(pos, sp.end + delta) }; if (sp.end > pos) return { ...sp, end: Math.max(sp.start, sp.end + delta) }; return sp; }).filter((sp) => sp.end > sp.start);
}

/* One movable text box. Drag to move, pinch (or scroll) to change the text
   size, tap without moving to edit. Width comes from the side bar. */
function TextBoxView({ bx, W, H, active, hit, onActivate, onMove, onScale, onTap, onDrop, fs, pop }) {
  const pts = React.useRef(new Map()); const lastDist = React.useRef(null); const moved = React.useRef(false);
  const dist = () => { const [a, b] = [...pts.current.values()]; return Math.hypot(a.x - b.x, a.y - b.y); };
  const h = {
    onPointerDown: (e) => { e.stopPropagation(); e.currentTarget.setPointerCapture(e.pointerId); pts.current.set(e.pointerId, { x: e.clientX, y: e.clientY }); if (pts.current.size === 1) moved.current = false; if (pts.current.size === 2) lastDist.current = dist(); },
    onPointerMove: (e) => { const p = pts.current.get(e.pointerId); if (!p) return; const n = { x: e.clientX, y: e.clientY }; pts.current.set(e.pointerId, n); if (pts.current.size >= 2) { const d = dist(); if (lastDist.current) onScale(d / lastDist.current); lastDist.current = d; moved.current = true; return; } const dx = (n.x - p.x) / W, dy = (n.y - p.y) / H; if (Math.abs(dx) + Math.abs(dy) > 0.002) moved.current = true; onMove(dx, dy); },
    onPointerUp: (e) => { pts.current.delete(e.pointerId); if (pts.current.size < 2) lastDist.current = null; if (pts.current.size === 0) { if (!moved.current) onTap(); else onDrop && onDrop(); } },
    onPointerCancel: (e) => { pts.current.delete(e.pointerId); lastDist.current = null; },
    onWheel: (e) => { onScale(1 - e.deltaY * 0.0025); },
  };
  return (
    <div {...h} onPointerDown={(e) => { onActivate(); h.onPointerDown(e); }} style={{ position: 'absolute', left: `${bx.box.x * 100}%`, top: `${bx.box.y * 100}%`, width: W * bx.box.w, transform: 'translate(-50%,-50%)', display: 'flex', justifyContent: bx.ts.align === 'left' ? 'flex-start' : bx.ts.align === 'right' ? 'flex-end' : 'center', cursor: 'grab', outline: hit ? '1.5px dashed #FF6B70' : 'none', outlineOffset: 6, borderRadius: 8, touchAction: 'none' }}>
      <span style={{ ...textBoxStyle(bx.ts, fs), animation: pop ? 'ss-place 520ms var(--ease-out) 260ms both' : 'none' }}>{runsOf(bx.text, bx.spans, bx.ts, fs)}</span>
    </div>
  );
}

  const { Icon } = CNS();
  const pts = React.useRef(new Map()); const lastDist = React.useRef(null); const moved = React.useRef(false); const raw = React.useRef(null);
  const dist = () => { const [a, b] = [...pts.current.values()]; return Math.hypot(a.x - b.x, a.y - b.y); };
  const h = {
    onPointerDown: (e) => { e.stopPropagation(); onActivate(); e.currentTarget.setPointerCapture(e.pointerId); pts.current.set(e.pointerId, { x: e.clientX, y: e.clientY }); if (pts.current.size === 1) { moved.current = false; raw.current = { x: pic.box.x, y: pic.box.y }; } if (pts.current.size === 2) lastDist.current = dist(); },
    onPointerMove: (e) => { const p = pts.current.get(e.pointerId); if (!p) return; const n = { x: e.clientX, y: e.clientY }; pts.current.set(e.pointerId, n); if (pts.current.size >= 2) { const d = dist(); if (lastDist.current) onScale(d / lastDist.current); lastDist.current = d; moved.current = true; return; } moved.current = true; raw.current = { x: raw.current.x + (n.x - p.x) / W, y: raw.current.y + (n.y - p.y) / H }; onMove(raw.current.x, raw.current.y); },
    onPointerUp: (e) => { pts.current.delete(e.pointerId); if (pts.current.size < 2) lastDist.current = null; if (pts.current.size === 0 && moved.current) onDrop(); },

    onWheel: (e) => onScale(1 - e.deltaY * 0.0025),

'1.5px dashed #FF6B70' : 'none', outlineOffset: 4 }} />
  );
}

                  <span style={{ display: 'inline-flex', alignItems: 'center', gap: 5, font: '700 12px var(--font-ui)', color: '#fff', whiteSpace: 'nowrap' }}>{done ? <Icon name="check" size={12} color="#4ADE80" /> : <span style={{ width: 12, height: 12, borderRadius: 99, border: '1.5px solid rgba(255,255,255,0.5)' }} />}{lbl}</span>
                  <span style={{ font: '500 11px var(--font-ui)', color: 'rgba(255,255,255,0.55)', whiteSpace: 'nowrap' }}>{s.role === 'cover' ? (done ? 'Title' : 'Write it') : (done ? 'Done' : 'Write it')}</span>
                </span> : null}
              </button>
            </React.Fragment>
          ); })}
          {onAddPoint && !slides.some((s) => s.role === 'cta') ? <button type="button" onClick={onAddPoint} aria-label="Add slide" style={{ width: 34, height: 60, flex: '0 0 auto', borderRadius: 6, border: '1.5px dashed rgba(255,255,255,0.35)', background: 'transparent', cursor: 'pointer', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}><Icon name="plus" size={16} color="#fff" /></button> : null}
        </div>
        {(() => { const cov = slides.find((s) => s.role === 'cover'), ct = slides.find((s) => s.role === 'cta'); const covOk = cov && cov.text && cov.text.trim(), ctOk = ct && ct.text && ct.text.trim(); const go = !covOk ? () => setSel(slides.indexOf(cov)) : !ctOk ? () => setSel(slides.indexOf(ct)) : onNext; const label = !covOk ? 'Cover' : !ctOk ? 'CTA' : 'Next'; const prim = covOk && ctOk; return (
          <button type="button" onClick={go} style={{ height: 44, padding: '0 16px', borderRadius: 999, border: prim ? 'none' : '1px solid rgba(255,255,255,0.25)', background: prim ? 'var(--blue-500)' : 'rgba(255,255,255,0.1)', color: '#fff', cursor: 'pointer', font: '700 15px var(--font-ui)', flex: '0 0 auto', display: 'inline-flex', alignItems: 'center', gap: 6, whiteSpace: 'nowrap' }}>{label}<Icon name="arrow-right" size={16} color="#fff" /></button>
        ); })()}
      </div>
      )}
      {media ? <MediaSheet onPhoto={(src) => { if (creator && !bg) { setBg(src); setHint(true); setTimeout(() => setHint(false), 2600); } else addPic(src); }} onInset={(id) => { patch({ inset: id }); setActive('inset'); }} onClose={() => setMedia(false)} mediaLabel={reel ? 'Screen recordings' : 'Screenshots'} /> : null}
    </div>
  );
}
