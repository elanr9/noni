const REWORD = {
  'Best clip first. Coaches decide in 10 seconds.': 'Your best clip goes first. Coaches decide in ten seconds.',
  'Keep it under 3 minutes. Nobody watches minute 4.': 'Three minutes max. Minute four never gets watched.',
  'Spot shadow yourself before every clip so they know who to watch.': 'Spot shadow yourself before each clip so they know who to follow.',
  'No music. Coaches mute it anyway.': 'Skip the music. Coaches mute it anyway.',
  'Put your name, grad year, position and email on the first frame.': 'Name, grad year, position and email on frame one.',
};
function creatorSlides(brief) {
  const cov = brief.find((s) => s.role === 'cover'), cta = brief.find((s) => s.role === 'cta');
  const pts = brief.filter((s) => s.role === 'point').map((s) => ({ ...s, text: REWORD[s.text] || s.text }));
  for (let i = pts.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [pts[i], pts[j]] = [pts[j], pts[i]]; }
  const at = 1 + Math.floor(Math.random() * Math.min(3, pts.length));
  const ctaNum = at + 1;
  pts.splice(at, 0, { ...cta, text: cta.text.replace(/^Tip \d+/, `Tip ${ctaNum}`) });
  return [cov, ...pts].map((s, i) => ({ ...s, id: i, bg: null, shot: s.shot }));
}

  const own = (id) => ({ id, role: 'point', text: '', own: true, bg: null, inset: null });

Math.max(1, post.count - 1) : 3; return [{ id: 0, role: 'cover', text: post.title, bg: null, inset: null }, ...Array.from({ length: k }).map((_, j) => own(j + 1)), { id: 99, role: 'cta', text: window.SS.story.cta.text, inset: window.SS.story.cta.inset, bg: null }]; }

function CreatorSlideshow({ label, scenario = 'morning' }) {
  const { TabBar } = RNS();
  const [tab, setTab] = React.useState('Home');
  const [posts, setPosts] = React.useState(() => scenario === 'reels' || scenario === 'editor' ? CREATOR_REELS : scenario === 'done' ? CREATOR_POSTS.map((p, i) => ({ ...p, status: i === 1 ? 'review' : 'posted' })) : scenario === 'one' ? CREATOR_POSTS.map((p, i) => ({ ...p, status: i === 0 ? 'posted' : i === 1 ? 'assigned' : 'review' })) : CREATOR_POSTS);
  const [screen, setScreen] = React.useState(() => scenario === 'editor' ? { name: 'make', post: CREATOR_REELS[0], startAt: 'edit' } : { name: 'home' });
  const [dark, setDark] = React.useState(scenario === 'editor');
  const [toast, setToast] = React.useState(null);
  const submit = () => { const post = screen.post; const nextPosts = posts.map((p) => p.id === post.id ? { ...p, status: 'review' } : p); setPosts(nextPosts); setDark(false); setScreen({ name: 'sent', post, left: nextPosts.filter((p) => p.status === 'assigned').length }); };
  let body;
  if (screen.name === 'home') body = <CreatorHome posts={posts} onCreate={(p) => setScreen({ name: 'make', post: p })} />;
  else if (screen.name === 'sent') body = <SentScreen post={screen.post} left={screen.left} onHome={() => setScreen({ name: 'home' })} />;
  else if (screen.post.format === 'reel') body = <CreatorReel startAt={screen.startAt} post={screen.post} onBack={() => { setScreen({ name: 'home' }); setDark(false); }} onSubmit={submit} onDark={setDark} />;
  else body = <CreatorHandheld post={screen.post} onBack={() => { setScreen({ name: 'home' }); setDark(false); }} onSubmit={submit} onDark={setDark} />;
  return (
    <Phone label={label} dark={dark}>
      <div key={screen.name} style={{ flex: 1, minHeight: 0, display: 'flex', flexDirection: 'column', animation: 'noni-screen-in var(--dur-base) var(--ease-out) both', position: 'relative' }}>{body}</div>
      {screen.name === 'home' ? <TabBar active={tab} onSelect={setTab} style={{ position: 'absolute', left: 16, right: 16, bottom: 22 }} items={[{ label: 'Home', icon: 'house', dot: posts.some((p) => p.status === 'assigned') }, { label: 'Posts', icon: 'layout-list' }, { label: 'Messages', icon: 'message-circle' }, { label: 'Analytics', icon: 'chart-column' }, { label: 'Profile', icon: 'circle-user-round' }]} /> : null}
      {toast ? <div style={{ position: 'absolute', left: 20, right: 20, bottom: 40, padding: '12px 16px', borderRadius: 16, background: 'var(--ink)', color: '#fff', font: '600 14px var(--font-ui)', boxShadow: 'var(--shadow-float)', zIndex: 90, display: 'flex', alignItems: 'center', gap: 10, animation: 'noni-toast-in var(--dur-base) var(--ease-out) both' }}><window.NoniDesignSystem_710e43.Icon name="send" size={16} color="#fff" />{toast}</div> : null}
    </Phone>
  );
}
