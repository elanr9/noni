/* TikTok-style reel editor. Preview up top, transport, a scrubbable
   timeline with the clip track, and one toolbar that swaps to clip tools
   when a clip is selected. */
const ENS = () => window.NoniDesignSystem_710e43;
const PX_PER_S = 56;

function ReelEditor({ slides, i, setI, patchAt, onBack, onNext, captions, setCaptions, onVoice, onAi, onAddClip, onDeleteClip, onRedoClip, onSplit, onTrim, cuts }) {
  const { Icon } = ENS();
  const W = 232, H = Math.round(W * 16 / 9);
  const durs = slides.map((s) => (s.clip ? s.clip.dur : 1));
  const total = durs.reduce((a, b) => a + b, 0);
  const starts = durs.map((_, k) => durs.slice(0, k).reduce((a, b) => a + b, 0));
  const [t, setT] = React.useState(0);
  const [playing, setPlaying] = React.useState(false);
  const [selClip, setSelClip] = React.useState(false);
  const [editText, setEditText] = React.useState(false);
  const [selText, setSelText] = React.useState(false);
  const [panel, setPanel] = React.useState(null);
  const [media, setMedia] = React.useState(false);
  const cl = (v, a, b) => Math.min(b, Math.max(a, v));
  const [tool, setTool] = React.useState(null);
  const [smooth, setSmooth] = React.useState(true);
  const [selKind, setSelKind] = React.useState(null); // 'clip' | 'text'
  const drag = React.useRef(null);
  React.useEffect(() => { if (!playing) return; const id = setInterval(() => setT((x) => { const n = x + 0.05; if (n >= total) { setPlaying(false); return total; } return n; }), 50); return () => clearInterval(id); }, [playing, total]);
  React.useEffect(() => { const k = starts.findIndex((s, j) => t >= s && t < s + durs[j]); if (k >= 0 && k !== i) setI(k); }, [t]);
  React.useEffect(() => { setSelText(false); setEditText(false); setPanel(null); }, [i]);
  const cur = slides[i] || slides[0];
  const ts = { ...DEFAULT_TSTYLE, ...(cur.tstyle || {}) };
  const fs = Math.round(W * 0.072 * (ts.size || 1));
  const L = layoutOf(cur);
  const fmt = (s) => `00:${String(Math.round(s)).padStart(2, '0')}`;
  const onDown = (e) => { drag.current = { x: e.clientX, t, moved: false, el: e.currentTarget }; e.currentTarget.setPointerCapture(e.pointerId); setPlaying(false); setSmooth(false); };
  const onMove = (e) => { if (!drag.current) return; if (Math.abs(e.clientX - drag.current.x) > 4) drag.current.moved = true; if (!drag.current.moved) return; const nt = Math.max(0, Math.min(total, drag.current.t - (e.clientX - drag.current.x) / PX_PER_S)); setT(nt); };
  const onUp = (e) => { const d = drag.current; drag.current = null; setSmooth(true); if (!d || d.moved) return; const r = d.el.getBoundingClientRect(); const cx = r.left + r.width / 2; const nt = Math.max(0, Math.min(total, d.t + (e.clientX - cx) / PX_PER_S)); setT(nt); const k = starts.findIndex((s, j) => nt >= s && nt < s + durs[j]); if (k < 0) return; setI(k); const y = e.clientY - r.top; if (y >= 60 && y <= 116) { setSelClip(true); setSelKind('clip'); setSelText(false); } else if (y >= 28 && y <= 54 && slides[k].text) { setSelClip(true); setSelKind('text'); setSelText(true); } else { setSelClip(false); setSelKind(null); setSelText(false); } };
  const jump = (k, sel = true) => { setSmooth(true); setT(starts[k]); setI(k); setSelClip(sel); setSelKind(sel ? 'clip' : null); setPlaying(false); };
  const Tb = ({ icon, l, on, dim, onClick }) => <button type="button" onClick={onClick} disabled={dim} style={{ flex: 1, display: 'flex', flexDirection: 'column', alignItems: 'center', gap: 5, padding: '8px 0', border: 'none', background: 'transparent', cursor: dim ? 'default' : 'pointer', color: on ? 'var(--blue-300)' : dim ? 'rgba(255,255,255,0.3)' : '#fff', font: '600 11px var(--font-ui)' }}><Icon name={icon} size={22} color={on ? 'var(--blue-300)' : dim ? 'rgba(255,255,255,0.3)' : '#fff'} />{l}</button>;
  const Ct = ({ icon, l, onClick }) => <button type="button" onClick={onClick} style={{ flex: '0 0 auto', width: 68, height: 68, display: 'flex', flexDirection: 'column', alignItems: 'center', justifyContent: 'center', gap: 6, borderRadius: 14, border: 'none', background: 'rgba(255,255,255,0.1)', cursor: 'pointer', color: '#fff', font: '600 11.5px var(--font-ui)' }}><Icon name={icon} size={21} color="#fff" />{l}</button>;
  return (
    <div style={{ position: 'absolute', top: -54, left: 0, right: 0, bottom: 0, zIndex: 70, background: '#000', display: 'flex', flexDirection: 'column', animation: 'ss-fade 240ms var(--ease-out) both' }}>
      <div style={{ height: 54, flexShrink: 0 }} />
      <div style={{ display: 'flex', alignItems: 'center', padding: '6px 12px 0' }}>
        <button type="button" onClick={onBack} aria-label="Back" style={{ width: 40, height: 40, borderRadius: 999, border: 'none', background: 'rgba(255,255,255,0.1)', cursor: 'pointer', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}><Icon name="chevron-left" size={22} color="#fff" /></button>
        <span style={{ flex: 1, textAlign: 'center', font: '700 13px var(--font-ui)', color: 'rgba(255,255,255,0.6)' }}>{cuts ? `Auto cut · ${cuts} pause${cuts === 1 ? '' : 's'} removed` : 'Auto cut on'}</span>
        <button type="button" onClick={onNext} aria-label="Next" style={{ width: 40, height: 40, borderRadius: 999, border: 'none', background: 'var(--blue-500)', cursor: 'pointer', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}><Icon name="arrow-right" size={22} color="#fff" /></button>
      </div>
      <div style={{ display: 'flex', justifyContent: 'center', padding: '10px 0 6px' }}>
        <div onClick={() => { if (!editText) setPlaying(!playing); }} style={{ position: 'relative', width: W, height: H, borderRadius: 10, overflow: 'hidden', background: photoBg(cur.bg ? cur.bg.src : 'camera'), cursor: 'pointer' }}>
          {(cur.pics || []).map((p) => <div key={p.id} onPointerDown={(ev) => ev.stopPropagation()} onClick={(ev) => ev.stopPropagation()} style={{ position: 'absolute', inset: 0, pointerEvents: 'none' }}><div style={{ position: 'absolute', inset: 0, pointerEvents: 'auto' }}><PicSticker pic={p} W={W} H={H} onActivate={() => {}} onMove={(rx, ry) => patchAt(i, { pics: (cur.pics || []).map((q) => q.id === p.id ? { ...q, box: { ...q.box, x: cl(rx, 0.05, 0.95), y: cl(ry, 0.05, 0.98) } } : q) })} onDrop={() => { if (p.box.y > 0.92) patchAt(i, { pics: (cur.pics || []).filter((q) => q.id !== p.id) }); }} onScale={(k) => patchAt(i, { pics: (cur.pics || []).map((q) => q.id === p.id ? { ...q, box: { ...q.box, w: cl(q.box.w * k, 0.15, 1) } } : q) })} /></div></div>)}
          {(cur.pics || []).length ? <span style={{ position: 'absolute', left: 0, right: 0, bottom: 6, display: 'flex', justifyContent: 'center', pointerEvents: 'none' }}><span style={{ padding: '3px 8px', borderRadius: 999, background: 'rgba(0,0,0,0.5)', color: '#fff', font: '600 10px var(--font-ui)' }}>Drag down to remove</span></span> : null}
          {cur.text && !editText ? <div onPointerDown={(ev) => ev.stopPropagation()} onClick={(ev) => ev.stopPropagation()} style={{ position: 'absolute', inset: 0 }}><TextBoxView bx={{ id: 'main', text: cur.text, spans: cur.spans || [], box: L.text, ts }} W={W} H={H} fs={fs} onActivate={() => {}} onTap={() => { setPlaying(false); setSelText(!selText); }} onMove={(dx, dy) => { setSelText(true); patchAt(i, { box: { ...L.text, x: cl(L.text.x + dx, 0.1, 0.9), y: cl(L.text.y + dy, 0.08, 0.92) } }); }} onScale={(k) => patchAt(i, { tstyle: { ...(cur.tstyle || {}), size: cl((ts.size || 1) * k, 0.5, 2.4) } })} />{selText ? <><span style={{ position: 'absolute', left: `${L.text.x * 100}%`, top: `${L.text.y * 100}%`, width: W * L.text.w + 16, height: fs * 1.2 * Math.max(1, Math.ceil(cur.text.length / Math.max(6, (L.text.w * W) / (fs * 0.52)))) + 16, transform: 'translate(-50%,-50%)', border: '1.5px dashed rgba(255,255,255,0.9)', borderRadius: 8, pointerEvents: 'none' }} /><button type="button" onClick={(ev) => { ev.stopPropagation(); setSelText(false); setEditText(true); }} style={{ position: 'absolute', left: `${L.text.x * 100}%`, top: `calc(${L.text.y * 100}% - ${fs * 1.2 * Math.max(1, Math.ceil(cur.text.length / Math.max(6, (L.text.w * W) / (fs * 0.52)))) / 2 + 30}px)`, transform: 'translate(-50%,-50%)', height: 30, padding: '0 12px', borderRadius: 999, border: 'none', background: '#fff', color: 'var(--ink)', cursor: 'pointer', font: '700 12.5px var(--font-ui)', display: 'inline-flex', alignItems: 'center', gap: 5, boxShadow: '0 2px 10px rgba(0,0,0,0.35)' }}><Icon name="pencil" size={12} color="var(--ink)" />Edit</button></> : null}</div> : null}
          {cur.inset ? <span style={{ position: 'absolute', left: `${L.inset.x * 100}%`, top: `${L.inset.y * 100}%`, transform: 'translate(-50%,-50%)', pointerEvents: 'none' }}><Shot item={window.SS.library.find((l) => l.id === cur.inset)} w={W * L.inset.w} /></span> : null}
          {editText ? (() => { const st = textBoxStyle(ts, fs); return <><span onClick={(ev) => ev.stopPropagation()} style={{ position: 'absolute', inset: 0, background: 'rgba(0,0,0,0.45)' }} /><div onClick={(ev) => ev.stopPropagation()} style={{ position: 'absolute', left: `${L.text.x * 100}%`, top: `${L.text.y * 100}%`, width: W * L.text.w, transform: 'translate(-50%,-50%)', display: 'flex', justifyContent: ts.align === 'left' ? 'flex-start' : ts.align === 'right' ? 'flex-end' : 'center' }}><span style={{ ...st, position: 'relative', width: '100%', display: 'block', minHeight: fs * 1.2 }}><span aria-hidden="true" style={{ display: 'block', whiteSpace: 'pre-wrap', wordBreak: 'break-word' }}>{cur.text}{'\u200b'}</span><textarea autoFocus rows={1} value={cur.text} onChange={(ev) => patchAt(i, { text: ev.target.value })} placeholder="Add text" style={{ position: 'absolute', inset: 0, width: '100%', height: '100%', margin: 0, padding: st.padding || 0, boxSizing: 'border-box', border: 'none', outline: 'none', resize: 'none', overflow: 'hidden', background: 'transparent', color: cur.text ? 'transparent' : 'rgba(255,255,255,0.6)', WebkitTextStroke: '0', textShadow: 'none', caretColor: '#1BA6EE', font: st.font, fontSize: fs, lineHeight: 1.2, letterSpacing: st.letterSpacing, textAlign: ts.align, whiteSpace: 'pre-wrap', wordBreak: 'break-word', display: 'block' }} /></span></div></>; })() : null}
          {captions && cur.caption ? <span style={{ position: 'absolute', left: '7%', right: '7%', top: '66%', display: 'flex', justifyContent: 'center', pointerEvents: 'none' }}><span style={{ font: `800 ${Math.round(W * 0.056)}px/1.25 var(--font-ui)`, color: '#fff', textAlign: 'center', WebkitTextStroke: '1px #000', paintOrder: 'stroke fill', textShadow: '0 1px 6px rgba(0,0,0,0.5)', display: '-webkit-box', WebkitLineClamp: 2, WebkitBoxOrient: 'vertical', overflow: 'hidden' }}>{cur.caption}</span></span> : null}
        </div>
      </div>
      {editText ? (() => { const setTs = (p) => patchAt(i, { tstyle: { ...(cur.tstyle || {}), ...p } }); return (
        <div style={{ display: 'flex', flexDirection: 'column', alignItems: 'center', gap: 8, padding: '4px 12px 8px' }}>
          {panel === 'font' ? <div style={{ display: 'flex', gap: 6 }}>{Object.entries(FONTS).map(([k, f]) => <button key={k} type="button" onClick={() => setTs({ font: k })} style={{ padding: '7px 12px', borderRadius: 10, border: 'none', background: ts.font === k ? '#fff' : 'rgba(255,255,255,0.12)', color: ts.font === k ? 'var(--ink)' : '#fff', cursor: 'pointer', font: f.font, fontSize: 13 }}>{f.label}</button>)}</div> : null}
          {panel === 'color' ? <div style={{ display: 'flex', gap: 8 }}>{SWATCHES.map((c) => <button key={c} type="button" onClick={() => setTs({ color: c })} aria-label={c} style={{ width: 26, height: 26, borderRadius: 999, border: '2.5px solid #fff', background: c, cursor: 'pointer', boxShadow: ts.color === c ? '0 0 0 2.5px #1BA6EE' : 'none' }} />)}</div> : null}
          <div style={{ display: 'flex', alignItems: 'center', gap: 4, padding: 6, borderRadius: 16, background: 'rgba(255,255,255,0.1)' }}>
            <button type="button" onClick={() => setPanel(panel === 'font' ? null : 'font')} style={{ width: 46, height: 40, borderRadius: 11, border: 'none', background: panel === 'font' ? 'rgba(255,255,255,0.25)' : 'transparent', color: '#fff', cursor: 'pointer', font: '800 17px var(--font-display)' }}>Aa</button>
            <button type="button" onClick={() => setPanel(panel === 'color' ? null : 'color')} aria-label="Color" style={{ width: 46, height: 40, borderRadius: 11, border: 'none', background: panel === 'color' ? 'rgba(255,255,255,0.25)' : 'transparent', cursor: 'pointer', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}><span style={{ width: 22, height: 22, borderRadius: 999, background: 'conic-gradient(#E5484D,#F6C445,#3DBE5B,#1BA6EE,#8E24AA,#E5484D)', border: '2px solid #fff' }} /></button>
            <button type="button" onClick={() => setTs({ bg: ts.bg === 'outline' ? 'solid' : ts.bg === 'solid' ? 'soft' : ts.bg === 'soft' ? 'none' : 'outline' })} aria-label="Text background" style={{ width: 46, height: 40, borderRadius: 11, border: 'none', background: 'transparent', cursor: 'pointer', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}><span style={{ width: 24, height: 24, borderRadius: 6, border: '2px solid #fff', background: ts.bg === 'solid' ? '#fff' : ts.bg === 'soft' ? 'rgba(255,255,255,0.45)' : 'transparent', display: 'inline-flex', alignItems: 'center', justifyContent: 'center', color: ts.bg === 'solid' ? 'var(--ink)' : '#fff', font: '800 13px var(--font-display)' }}>A</span></button>
            <button type="button" onClick={() => setTs({ align: ts.align === 'center' ? 'left' : ts.align === 'left' ? 'right' : 'center' })} aria-label="Align" style={{ width: 46, height: 40, borderRadius: 11, border: 'none', background: 'transparent', cursor: 'pointer', display: 'inline-flex', flexDirection: 'column', alignItems: ts.align === 'left' ? 'flex-start' : ts.align === 'right' ? 'flex-end' : 'center', justifyContent: 'center', gap: 3, padding: '0 11px' }}>{[22, 14, 18].map((w, k) => <span key={k} style={{ width: w, height: 2.5, borderRadius: 2, background: '#fff' }} />)}</button>
            <button type="button" onClick={() => { setEditText(false); setPanel(null); }} style={{ height: 40, padding: '0 16px', borderRadius: 11, border: 'none', background: '#fff', color: 'var(--ink)', cursor: 'pointer', font: '700 14px var(--font-ui)' }}>Done</button>
          </div>
        </div>); })() : null}
      <div style={{ display: editText ? 'none' : 'flex', alignItems: 'center', padding: '2px 16px', gap: 14 }}>
        <span style={{ font: '600 13px var(--font-ui)', color: '#fff', fontVariantNumeric: 'tabular-nums' }}>{fmt(t)}<span style={{ color: 'rgba(255,255,255,0.5)' }}>/{fmt(total)}</span></span>
        <span style={{ flex: 1, display: 'flex', justifyContent: 'center' }}><button type="button" onClick={() => { if (t >= total) setT(0); setPlaying(!playing); }} aria-label={playing ? 'Pause' : 'Play'} style={{ width: 36, height: 36, border: 'none', background: 'transparent', cursor: 'pointer', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}><Icon name={playing ? 'pause' : 'play'} size={22} color="#fff" /></button></span>
        <Icon name="rotate-ccw" size={18} color="rgba(255,255,255,0.4)" /><Icon name="arrow-left-right" size={18} color="rgba(255,255,255,0.4)" />
      </div>

        <div style={{ position: 'absolute', left: 0, right: 0, top: 0, height: 20, overflow: 'hidden' }}><div style={{ position: 'absolute', left: `calc(50% - ${t * PX_PER_S}px)`, display: 'flex', transition: smooth && !playing ?       )}
      {media ? <MediaSheet onPhoto={(src) => { const id = 'p' + Date.now(); patchAt(i, { pics: [...(cur.pics || []), { id, src, box: { x: 0.5, y: 0.26, w: 0.44 }, boxIg: { x: 0.5, y: 0.3, w: 0.44 } }] }); setMedia(false); }} onInset={(id) => { patchAt(i, { inset: id }); setMedia(false); }} onClose={() => setMedia(false)} mediaLabel="Inkbound" /> : null}
      {tool === 'trim' && cur.clip ? <div style={{ position: 'absolute', left: 12, right: 12, bottom: 100, zIndex: 5, display: 'flex', alignItems: 'center', gap: 12, padding: '10px 14px', borderRadius: 14, background: 'rgba(20,24,30,0.92)', animation: 'ss-rise 200ms var(--ease-out) both' }}><span style={{ flex: 1, font: '700 13px var(--font-ui)', color: '#fff' }}>Clip length</span><button type="button" onClick={() => onTrim(i, Math.max(1, cur.clip.dur - 1))} aria-label="Shorter" style={{ width: 36, height: 36, borderRadius: 999, border: '1px solid rgba(255,255,255,0.3)', background: 'transparent', color: '#fff', cursor: 'pointer', font: '600 20px var(--font-ui)' }}>−</button><span style={{ minWidth: 36, textAlign: 'center', font: '800 18px var(--font-display)', color: '#fff' }}>{cur.clip.dur}s</span><button type="button" onClick={() => onTrim(i, Math.min(60, cur.clip.dur + 1))} aria-label="Longer" style={{ width: 36, height: 36, borderRadius: 999, border: '1px solid rgba(255,255,255,0.3)', background: 'transparent', color: '#fff', cursor: 'pointer', font: '600 20px var(--font-ui)' }}>+</button><button type="button" onClick={() => setTool(null)} style={{ height: 36, padding: '0 14px', borderRadius: 999, border: 'none', background: '#fff', color: 'var(--ink)', cursor: 'pointer', font: '700 13px var(--font-ui)' }}>Done</button></div> : null}
    </div>
  );
}
