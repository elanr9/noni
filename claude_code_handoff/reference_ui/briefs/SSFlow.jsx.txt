/* Title to feature, by meaning, then the CTA shaped to the post type so it reads
   as the last slide of that post, not an ad. Stand-in for the backend matcher. */
function matchFeature(title, postType, n, forceId) {
  const t = (title || '').toLowerCase(), L = window.SS.library, by = (id) => L.find((l) => l.id === id);
  const rules = [[/highlight|film|clip|video|footage/, 'hl'], [/reply|respond|message|text back|follow.?up/, 'reply'], [/inbox|unread|dm/, 'inbox'], [/email|outreach|reach out|cold|campaign/, 'sent'], [/which school|find school|search|programs? (i|you) can|d1|d2|d3|division|recruited/, 'search'], [/one school|chance|acceptance|cost|tuition|scholarship/, 'school'], [/plan|priorit|fit|matters/, 'bestfit']];
  let it = by('general');
  for (const [re, id] of rules) if (re.test(t)) { it = by(id); break; }
  if (forceId) it = by(forceId) || it;
  const hook = { hl: 'Inkbound makes your highlight video in 6 minutes.', reply: 'Inkbound drafts every coach reply for you.', inbox: 'Inkbound puts every coach reply in one inbox.', sent: 'Inkbound emails 50 coaches at once.', search: 'Inkbound finds the schools you can actually get into.', school: 'Inkbound shows your real chance at every school.', bestfit: 'Inkbound builds your plan from what matters to you.' }[it.id];
  const type = postType && postType.id;
  const num = n ? `${n}` : null;
  const shaped = !hook ? it.cta
    : type === 'howto' ? `${num ? `Step ${num}` : 'Last step'}: ${hook}`
    : type === 'tips' ? `${num ? `Tip ${num}` : 'Last tip'}: ${hook}`
    : type === 'start' ? `${num ? `Step ${num}` : 'Start here'}: ${hook}`
    : type === 'myths' ? `${num ? `Myth ${num}, busted` : 'Truth'}: ${hook}`
    : type === 'storytime' ? `Then I found Inkbound. ${hook.replace(/^Inkbound /, 'It ')}`
    : (type === 'talking' || type === 'explainer' || type === 'green' || type === 'longtext') ? hook
    : it.cta;
  return { ...it, cta: shaped };
}

                <span style={{ font: '700 11px var(--font-ui)', letterSpacing: '0.04em', textTransform: 'uppercase', color: c ? 'var(--blue-300)' : 'rgba(255,255,255,0.5)' }}>{c ? 'CTA \u00b7 lands anywhere per creator' : cov ? 'Hook \u00b7 script below, word for word' : cr ? 'Creator records' : story ? `Beat ${k}` : 'Point \u00b7 order shuffled per creator'}</span>
                <span style={{ font: `${cov ? 700 : 600} 15px/1.35 var(--font-ui)`, color: cr ? 'rgba(255,255,255,0.6)' : '#fff', textWrap: 'pretty' }}>{r.text}</span>
                {r.note ? <span style={{ font: '500 12.5px var(--font-ui)', color: 'rgba(255,255,255,0.5)' }}>{r.note}</span> : null}
              </span>
              {it ? <span style={{ flex: '0 0 auto', alignSelf: 'center' }}><Shot item={it} w={44} /></span> : null}
            </div>
          ); })}
          {caption ? <div style={{ marginTop: 6, padding: '12px 14px', borderRadius: 14, background: 'rgba(255,255,255,0.06)', display: 'flex', flexDirection: 'column', gap: 4 }}><span style={{ font: '700 11px var(--font-ui)', letterSpacing: '0.04em', textTransform: 'uppercase', color: 'rgba(255,255,255,0.5)' }}>Caption</span><span style={{ font: '500 14px/1.4 var(--font-ui)', color: '#fff' }}>{caption} <span style={{ fontWeight: 700 }}>{tags}</span></span></div> : null}
        </div>
        <BackBtn /><Tabs dark />
      </div>
      ) : (
      <div style={{ position: 'relative', width: W, height: H, borderRadius: 22, overflow: 'hidden', flexShrink: 0 }}>
        <div onPointerDown={onDown} onPointerMove={onMove} onPointerUp={onUp} onPointerCancel={onUp} onWheel={onWheel} style={{ display: 'flex', width: '100%', height: '100%', touchAction: 'pan-y', cursor: 'grab', transform: `translateX(calc(${-i * 100}% + ${dx}px))`, transition: drag.current ? 'none' : 'transform 320ms var(--ease-out)' }}>
          {items.map((s, k) => <div key={s.id ?? k} style={{ width: '100%', height: '100%', flex: '0 0 100%', background: s.who === 'creator' ? '#1C2128' : 'linear-gradient(180deg,#2A3038,#1C2128)' }}><SlideCard w={W} text={s.text} inset={s.inset} box={s.box} insetBox={s.insetBox} photo={s.bg ? s.bg.src : (s.photo || (s.who === 'creator' ? null : 'camera'))} bgT={s.bg ? s.bg.tiktok : undefined} tstyle={s.tstyle} extras={s.extras} spans={s.spans} who={s.who || 'mixed'} label={s.label} bare />{s.caption ? <span style={{ position: 'absolute', left: '6%', right: '20%', top: '58%', display: 'flex', justifyContent: 'center', pointerEvents: 'none' }}><span style={{ display: 'inline-block', padding: '6px 10px', borderRadius: 8, background: 'rgba(0,0,0,0.55)', color: '#fff', font: '800 17px/1.25 var(--font-ui)', textAlign: 'center', textWrap: 'pretty' }}>{s.caption}</span></span> : null}</div>)}
        </div>
        <FypChrome caption={caption} tags={tags} w={W} index={i} count={items.length} />
        {reel ? <button type="button" onClick={() => { if (!playing && i >= items.length - 1) setI(0); setPlaying(!playing); }} aria-label={playing ? 'Pause' : 'Play'} style={{ position: 'absolute', left: 58, top: 10, width: 40, height: 40, borderRadius: 999, border: 'none', background: playing ? '#fff' : 'rgba(20,24,30,0.55)', cursor: 'pointer', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}><Icon name={playing ? 'pause' : 'play'} size={18} color={playing ? 'var(--ink)' : '#fff'} /></button> : null}
        {reel ? <div style={{ position: 'absolute', left: 12, right: 12, bottom: 6, height: 3, borderRadius: 99, background: 'rgba(255,255,255,0.25)' }}><div style={{ width: `${((i + 1) / items.length) * 100}%`, height: '100%', borderRadius: 99, background: '#fff', transition: 'width 320ms var(--ease-out)' }} /></div> : null}
        {reel ? <Tabs /> : null}
        <span style={{ position: 'absolute', top: reel ? 56 : 54, right: 14, padding: '3px 9px', borderRadius: 999, background: 'rgba(0,0,0,0.5)', color: '#fff', font: '700 12px var(--font-ui)', whiteSpace: 'nowrap' }}>{reel ? `Clip ${i + 1} of ${items.length}` : `${i + 1} / ${items.length}`}</span>
        {items[i]?.role === 'cta' || (!story && items[i]?.inset) ? <span style={{ position: 'absolute', top: reel ? 56 : 54, left: reel ? 14 : 60, display: 'inline-flex', alignItems: 'center', gap: 5, padding: '3px 9px', borderRadius: 999, background: 'var(--blue-500)', color: '#fff', font: '700 12px var(--font-ui)' }}><Icon name="zap" size={12} color="#fff" />{creator ? 'Your plug slide' : 'CTA · lands anywhere, number refilled per creator'}</span> : null}
        {!story && !creator && items[i]?.role === 'point' ? <span style={{ position: 'absolute', top: reel ? 56 : 54, left: reel ? 14 : 60, display: 'inline-flex', alignItems: 'center', gap: 5, padding: '3px 9px', borderRadius: 999, background: 'rgba(0,0,0,0.5)', color: '#fff', font: '700 12px var(--font-ui)' }}><Icon name="arrow-left-right" size={12} color="#fff" />Order shuffled per creator</span> : null}
        {story && items[i]?.who === 'creator' ? <span style={{ position: 'absolute', top: reel ? 56 : 54, left: reel ? 14 : 60, display: 'inline-flex', alignItems: 'center', gap: 5, padding: '3px 9px', borderRadius: 999, background: 'rgba(0,0,0,0.5)', color: '#fff', font: '700 12px var(--font-ui)' }}><Icon name={reel ? 'video' : 'camera'} size={12} color="#fff" />{openCount ? 'Creator picks the count' : `Creator records ${gap.min}`}</span> : null}
        <BackBtn />
        {creator && onEdit ? <button type="button" onClick={() => onEdit(i)} style={{ position: 'absolute', top: 10, right: 12, height: 40, padding: '0 14px 0 12px', borderRadius: 999, border: 'none', background: '#fff', color: 'var(--ink)', cursor: 'pointer', font: '700 14px var(--font-ui)', display: 'inline-flex', alignItems: 'center', gap: 6, boxShadow: '0 4px 16px rgba(0,0,0,0.3)' }}><Icon name="pencil" size={15} color="var(--ink)" />Edit</button> : null}
      </div>
      )}
      {popup ? <div style={{ flex: 1, minHeight: 0, display: 'flex', alignItems: 'center', justifyContent: 'center', padding: '4px 14px 6px' }}><span style={{ font: '600 13px var(--font-ui)', color: 'rgba(255,255,255,0.6)' }}>Preview</span></div> : creator ? <div style={{ flex: 1, minHeight: 0, display: 'flex', alignItems: 'center', gap: 10, padding: '4px 14px 6px' }}>
        <div style={{ flex: 1, minWidth: 0, display: 'flex', alignItems: 'center', gap: 5, overflowX: 'auto', scrollbarWidth: 'none', padding: '3px 2px' }}>{items.map((s, k) => <button key={s.id ?? k} type="button" onClick={() => setI(k)} aria-label={`Slide ${k + 1}`} style={{ width: 26, height: 46, flex: '0 0 auto', padding: 0, borderRadius: 6, border: 'none', cursor: 'pointer', position: 'relative', overflow: 'hidden', background: s.bg ? photoBg(s.bg.src) : '#2A3038', boxShadow: k === i ? '0 0 0 2px #fff' : 'none', opacity: k === i ? 1 : 0.7, transition: 'box-shadow 160ms, opacity 160ms' }}>{s.role === 'cta' ? <span style={{ position: 'absolute', right: 2, top: 2, width: 10, height: 10, borderRadius: 999, background: 'var(--blue-500)' }} /> : null}</button>)}</div>
        <button type="button" onClick={onSave} style={{ height: 42, padding: '0 16px', borderRadius: 999, border: 'none', background: 'var(--blue-500)', color: '#fff', cursor: 'pointer', font: '700 15px var(--font-ui)', flex: '0 0 auto', display: 'inline-flex', alignItems: 'center', gap: 6, whiteSpace: 'nowrap' }}>{saveLabel}<Icon name={saveIcon} size={16} color="#fff" /></button>
      </div> : <div style={{ flex: 1, minHeight: 0, display: 'flex', alignItems: 'center', gap: 10, padding: '4px 14px 6px' }}>
        <div style={{ flex: 1, minWidth: 0, overflowX: 'auto', scrollbarWidth: 'none' }}><StepStrip steps={steps} current={current} dark onJump={onJump} /></div>
        <button type="button" onClick={onSave} style={{ height: 42, padding: '0 16px', borderRadius: 999, border: 'none', background: 'var(--blue-500)', color: '#fff', cursor: 'pointer', font: '700 15px var(--font-ui)', flex: '0 0 auto', display: 'inline-flex', alignItems: 'center', gap: 6, whiteSpace: 'nowrap' }}>{saveLabel}<Icon name={saveIcon} size={16} color="#fff" /></button>
      </div>}
    </div>
  );
}
