/* Freestyle, creator side. Pick photos like TikTok (multi-select, numbered),
   then edit with a slide strip, text tools and an AI helper for ideas. */
const FNS2 = () => window.NoniDesignSystem_710e43;

function FreestylePicker({ onDone, onClose, min = 1, title = 'Select photos' }) {
  const { Icon } = FNS2();
  const [sel, setSel] = React.useState([]);
  const toggle = (id) => setSel((a) => a.includes(id) ? a.filter((x) => x !== id) : a.length >= 10 ? a : [...a, id]);
  return (
    <div style={{ position: 'absolute', top: -54, left: 0, right: 0, bottom: 0, zIndex: 80, background: '#000', display: 'flex', flexDirection: 'column', animation: 'ss-rise 260ms var(--ease-out) both' }}>
      <div style={{ height: 54, flexShrink: 0 }} />
      <div style={{ display: 'flex', alignItems: 'center', padding: '6px 14px 10px', gap: 10 }}>
        <button type="button" onClick={onClose} aria-label="Close" style={{ width: 40, height: 40, borderRadius: 999, border: 'none', background: 'rgba(255,255,255,0.1)', cursor: 'pointer', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}><Icon name="x" size={20} color="#fff" /></button>
        <span style={{ flex: 1, textAlign: 'center', font: '700 16px var(--font-ui)', color: '#fff' }}>{title}</span>
        <span style={{ width: 40 }} />
      </div>
      <div style={{ flex: 1, minHeight: 0, overflowY: 'auto', padding: '0 2px', display: 'grid', gridTemplateColumns: 'repeat(3, 1fr)', gap: 2, alignContent: 'start' }}>
        <button type="button" onClick={() => { setSel((a) => [...a, 'camera']); }} style={{ aspectRatio: '3 / 4', border: 'none', background: '#1C2128', cursor: 'pointer', display: 'flex', flexDirection: 'column', alignItems: 'center', justifyContent: 'center', gap: 6, color: '#fff', font: '700 12px var(--font-ui)' }}><Icon name="camera" size={26} color="#fff" />Camera</button>
        {ROLL.map((r) => { const k = sel.indexOf(r.id); return (
          <button key={r.id} type="button" onClick={() => toggle(r.id)} style={{ position: 'relative', aspectRatio: '3 / 4', border: 'none', background: r.bg, cursor: 'pointer', padding: 0, opacity: sel.length && k < 0 ? 0.75 : 1 }}>
            <span style={{ position: 'absolute', top: 6, right: 6, width: 24, height: 24, borderRadius: 999, border: '2px solid #fff', background: k >= 0 ? 'var(--blue-500)' : 'rgba(0,0,0,0.25)', color: '#fff', font: '800 12px var(--font-ui)', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}>{k >= 0 ? k + 1 : ''}</span>
          </button>
        ); })}
      </div>
      <div style={{ padding: '12px 14px 30px', display: 'flex', alignItems: 'center', gap: 10 }}>
        <div style={{ flex: 1, display: 'flex', gap: 6, overflowX: 'auto', scrollbarWidth: 'none' }}>{sel.map((id, k) => <span key={id + k} style={{ width: 34, height: 46, flex: '0 0 auto', borderRadius: 6, background: photoBg(id), position: 'relative' }}><button type="button" onClick={() => setSel((a) => a.filter((_, j) => j !== k))} aria-label="Remove" style={{ position: 'absolute', top: -5, right: -5, width: 18, height: 18, borderRadius: 999, border: 'none', background: '#fff', cursor: 'pointer', display: 'inline-flex', alignItems: 'center', justifyContent: 'center', padding: 0 }}><Icon name="x" size={11} color="var(--ink)" /></button></span>)}</div>
        <button type="button" disabled={sel.length < min} onClick={() => onDone(sel)} style={{ height: 46, padding: '0 20px', borderRadius: 999, border: 'none', background: sel.length >= min ? 'var(--blue-500)' : 'rgba(255,255,255,0.12)', color: sel.length >= min ? '#fff' : 'rgba(255,255,255,0.4)', cursor: sel.length >= min ? 'pointer' : 'default', font: '700 15px var(--font-ui)', whiteSpace: 'nowrap' }}>Next{sel.length ? ` (${sel.length})` : ''}</button>
      </div>
    </div>
  );
}

/* AI helper. Quick asks or a free question; answers come back as cards you can apply. */
const AI_HOOKS = ['Nobody tells you this about recruiting', '5 things I wish I knew before my first showcase', 'POV: a D1 coach just opened your email'];
const AI_LINES = ['Film the whole play, not just the finish', 'Email 5 coaches a week, every week', 'Your grad year goes in the subject line', 'Coaches reply to questions, not life stories', 'One highlight video. Under 3 minutes.', 'Update your film every season'];
function AiSheet({ slides, i, onApply, onApplyAll, onAddSlides, onClose, reel = false, onCut }) {
  const { Icon } = FNS2();
  const [q, setQ] = React.useState('');
  const [busy, setBusy] = React.useState(false);
  const [res, setRes] = React.useState(null); // { kind, title, items }
  const ask = (kind, label) => { setBusy(true); setRes(null); setTimeout(() => { setBusy(false); if (kind === 'hook') setRes({ kind, title: 'Hooks for your cover', items: AI_HOOKS }); else if (kind === 'line') setRes({ kind, title: `Text for slide ${i + 1}`, items: AI_LINES.slice(i % 3, i % 3 + 3) }); else if (kind === 'all') setRes({ kind, title: 'Text for every slide', items: slides.filter((s) => s.own).map((s, k) => k === 0 ? AI_HOOKS[0] : AI_LINES[(k - 1) % AI_LINES.length]) }); else if (kind === 'count') setRes({ kind, title: 'How many slides', items: [`You have ${slides.filter((s) => s.own).length}. Recruiting tips do best at 6 to 8. Add ${Math.max(0, 6 - slides.filter((s) => s.own).length)} more, one idea each.`] }); else setRes({ kind: 'free', title: label, items: ['Lead with the number. Put the strongest tip on slide 2, not last.', 'Keep every line under 9 words so it reads in one glance.'] }); }, 1100);
  };
  const Chip = ({ k, l, icon }) => <button type="button" onClick={() => ask(k, l)} style={{ display: 'inline-flex', alignItems: 'center', gap: 6, height: 38, padding: '0 14px', borderRadius: 999, border: '1px solid var(--border)', background: 'var(--white)', color: 'var(--ink)', cursor: 'pointer', font: '600 13.5px var(--font-ui)', whiteSpace: 'nowrap', flex: '0 0 auto' }}><Icon name={icon} size={14} color="var(--blue-500)" />{l}</button>;
  return (
    <SSheet onClose={onClose} height="72%">
      {(dismiss) => (<>
        <div style={{ padding: '8px 20px 12px', display: 'flex', alignItems: 'center', gap: 10 }}>
          <span style={{ width: 36, height: 36, borderRadius: 999, background: 'var(--blue-500)', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}><Icon name="sparkles" size={18} color="#fff" /></span>
          <span style={{ flex: 1, font: '700 18px var(--font-display)', letterSpacing: '-0.3px', color: 'var(--ink)' }}>Ask AI</span>
          <button type="button" onClick={() => dismiss()} aria-label="Close" style={{ width: 34, height: 34, borderRadius: 999, border: 'none', background: 'var(--fill-quiet)', cursor: 'pointer', display: 'inline-flex', alignItems: 'center', justifyContent: 'center' }}><Icon name="x" size={17} color="var(--slate-500)" /></button>
        </div>
        <div style={{ display: 'flex', gap: 8, overflowX: 'auto', scrollbarWidth: 'none', padding: '0 20px 12px' }}>
          {reel && onCut ? <button type="button" onClick={() => dismiss(onCut)} style={{ display: 'inline-flex', alignItems: 'center', gap: 6, height: 38, padding: '0 14px', borderRadius: 999, border: 'none', background: 'var(--blue-500)', color: '#fff', cursor: 'pointer', font: '700 13.5px var(--font-ui)', whiteSpace: 'nowrap', flex: '0 0 auto' }}><Icon name="zap" size={14} color="#fff" />Cut the dead air</button> : null}<Chip k="hook" l={reel ? 'Hook ideas' : 'Hook for the cover'} icon="zap" /><Chip k="line" l={reel ? 'Rewrite my line' : 'Text for this slide'} icon="pencil" />{reel ? null : <Chip k="all" l="Text for all slides" icon="layout-list" />}<Chip k="count" l={reel ? 'Is it too long?' : 'How many slides?'} icon="images" />
        </div>
        <div style={{ flex: 1, minHeight: 0, overflowY: 'auto', padding: '0 20px', display: 'flex', flexDirection: 'column', gap: 8 }}>
          {busy ? <div style={{ display: 'flex', alignItems: 'center', gap: 10, padding: '14px 16px', borderRadius: 'var(--radius-md)', background: 'var(--blue-50)', font: '600 14px var(--font-ui)', color: 'var(--blue-700)' }}><span style={{ width: 16, height: 16, borderRadius: 999, border: '2.5px solid var(--blue-200)', borderTopColor: 'var(--blue-500)', animation: 'ss-spin 0.8s linear infinite' }} />Looking at your slides</div> : null}
          {res ? <>
            <SLabel>{res.title}</SLabel>
            {res.items.map((t, k) => <div key={k} style={{ display: 'flex', alignItems: 'center', gap: 10, padding: '12px 14px', borderRadius: 'var(--radius-md)', background: 'var(--white)', border: '1px solid var(--border)' }}>
              {res.kind === 'all' ? <span style={{ width: 22, height: 22, borderRadius: 999, background: 'var(--fill-quiet)', font: '800 11px var(--font-ui)', color: 'var(--slate-500)', display: 'inline-flex', alignItems: 'center', justifyContent: 'center', flex: '0 0 auto' }}>{k + 1}</span> : null}
              <span style={{ flex: 1, font: '500 15px/1.35 var(--font-ui)', color: 'var(--ink)', textWrap: 'pretty' }}>{t}</span>
              {res.kind === 'hook' || res.kind === 'line' ? <button type="button" onClick={() => dismiss(() => onApply(res.kind === 'hook' ? 0 : i, t))} style={{ height: 32, padding: '0 12px', borderRadius: 999, border: 'none', background: 'var(--blue-500)', color: '#fff', cursor: 'pointer', font: '700 13px var(--font-ui)', flex: '0 0 auto' }}>Use</button> : null}
            </div>)}
            {res.kind === 'all' ? <SButton onClick={() => dismiss(() => onApplyAll(res.items))}>Put these on my slides</SButton> : null}
            {res.kind === 'count' && onAddSlides ? <SButton variant="secondary" icon="image-plus" onClick={() => dismiss(() => onAddSlides())}>Add more photos</SButton> : null}
          </> : null}
          {!res && !busy ? <span style={{ padding: '18px 4px', font: '500 14px/1.45 var(--font-ui)', color: 'var(--slate-500)' }}>Pick a quick ask above, or type what you need.</span> : null}
        </div>
        <div style={{ display: 'flex', gap: 8, padding: '12px 20px 0' }}>
          <SField value={q} onChange={setQ} placeholder="Ask for anything" style={{ flex: 1, padding: '11px 16px' }} />
          <button type="button" disabled={!q.trim()} onClick={() => { ask('free', q.trim()); setQ(''); }} aria-label="Send" style={{ width: 50, height: 50, borderRadius: 999, border: 'none', background: q.trim() ? 'var(--blue-500)' : 'var(--fill-quiet)', cursor: 'pointer', display: 'inline-flex', alignItems: 'center', justifyContent: 'center', flex: '0 0 auto' }}><Icon name="send" size={18} color={q.trim() ? '#fff' : 'var(--slate-400)'} /></button>
        </div>
      </>)}
    </SSheet>
  );
}
Object.assign(window, { FreestylePicker, AiSheet });
