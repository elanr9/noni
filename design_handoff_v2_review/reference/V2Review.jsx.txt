/* Review tab: one card per creator, creator queue, post review, creator profile, thread, music and account approvals. */
const reviewStore = { queue: window.V2.queue.slice(), music: window.V2.music.slice(), accounts: window.V2.accounts.slice(), busy: false, listeners: new Set() };
const setReview = (patch) => { Object.assign(reviewStore, patch); reviewStore.listeners.forEach((l) => l({})); };
window.setBusy = (on) => setReview({ busy: on, queue: on ? window.V2.queue.concat(window.V2.busyQueue) : window.V2.queue.slice() });
const useReview = () => { const [, tick] = React.useState({}); React.useEffect(() => { reviewStore.listeners.add(tick); return () => reviewStore.listeners.delete(tick); }, []); return reviewStore; };
const useLoaded = (ms = 500) => { const [l, setL] = React.useState(false); React.useEffect(() => { const t = setTimeout(() => setL(true), ms); return () => clearTimeout(t); }, []); return l; };

const statusLine = (posts) => {
  const waiting = posts.filter((p) => p.state === 'Waiting').length, retakes = posts.filter((p) => p.take > 1).length;
  const parts = []; if (waiting) parts.push(`${waiting} waiting`); if (retakes) parts.push(`${retakes} take 2`); if (!waiting && posts.length) parts.push(`${posts.length} back after changes`);
  return parts.join(', ');
};

const countLine = (posts) => `${posts.length} ${posts.length === 1 ? 'post' : 'posts'}`;

function CreatorCard({ c, posts, onClick, onPost }) {
  return (
    <Card onClick={onClick} pad={'14px 16px'} style={{ display: 'flex', alignItems: 'center', gap: 14 }}>
      <Avatar name={c.name} photo={c.photo} tone={c.tone} size={64} />
      <span className="t-card" style={{ flex: 1, minWidth: 0, whiteSpace: 'nowrap', fontSize: 22, lineHeight: '26px' }}>{c.name}</span>
      <span style={{ display: 'flex', flexDirection: 'column', alignItems: 'flex-end', gap: 6, flex: '0 0 auto' }}>
        <span style={{ whiteSpace: 'nowrap', font: '600 13px/18px var(--font)', color: 'var(--blue-deep)' }}>{countLine(posts)}</span>
        <span style={{ display: 'flex', gap: 5, justifyContent: 'flex-end', maxWidth: 160, flexWrap: 'wrap' }}>
          {posts.map((p) => <span key={p.id} role="button" className="p" onClick={(e) => { e.stopPropagation(); onPost(p); }} style={{ display: 'block' }}><Thumb w={26} post={p} title={false} radius={6} /></span>)}
        </span>
      </span>
      <Icon name="chevron_right" size={20} color="var(--grey-3)" />
    </Card>
  );
}

function ReviewScreen() {
  const nav = useNav();
  const { queue, music, accounts, busy } = useReview();
  const loaded = useLoaded();
  const [lane, setLane] = React.useState('Posts');
  const total = queue.length + music.length + accounts.length;
  const groups = window.V2.creators.concat(busy ? window.V2.extraCreators : []).map((c) => ({ c, posts: queue.filter((p) => p.creator === c.id) })).filter((g) => g.posts.length);
  return (
    <Screen title="Review" sub={loaded ? (total ? <Chip tone="blue" icon="schedule">{total} waiting</Chip> : <Chip tone="green" icon="check">All clear</Chip>) : <span style={{ height: 26 }} />} right={null}>
      <Seg options={[{ label: 'Posts', count: queue.length, tone: 'blue' }, { label: 'Music', count: music.length, tone: 'purple' }, { label: 'Accounts', count: accounts.length, tone: 'amber' }]} value={lane} onChange={setLane} style={{ marginBottom: 16 }} />
      {!loaded ? [0, 1, 2, 3].map((i) => <Skeleton key={i} h={76} />) : null}
      {loaded && lane === 'Posts' ? (groups.length === 0 ? <Empty>Nothing to review.</Empty> : groups.map((g) => <CreatorCard key={g.c.id} c={g.c} posts={g.posts} onClick={() => nav.push('CreatorQueue', { id: g.c.id })} onPost={(p) => nav.push('ReviewDetail', { id: p.id })} />)) : null}

        <Card key={m.id} onClick={() => nav.push('Music', { id: m.id })} style={{ display: 'flex', alignItems: 'center', gap: 14 }}>

        </Card>
      ); })) : null}

  );
}
