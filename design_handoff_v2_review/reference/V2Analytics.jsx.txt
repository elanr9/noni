/* Analytics tab and settings. */
function Bars({ days, unit }) {
  const max = Math.max(...days.map((d) => d.v));
  const [sel, setSel] = React.useState(3);
  return (
    <div>
      <div style={{ display: 'flex', alignItems: 'baseline', gap: 8, paddingBottom: 12 }}><span className="t-num-s">{days[sel].v}k</span><span className="t-meta" style={{ whiteSpace: 'nowrap' }}>{unit} on {['Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday', 'Sunday'][sel]}</span></div>
      <div style={{ display: 'grid', gridTemplateColumns: 'repeat(7, 1fr)', gap: 8, alignItems: 'end', height: 120 }}>
        {days.map((d, i) => <button key={d.d} type="button" className="p" onClick={() => { haptic(); setSel(i); }} style={{ height: `${(d.v / max) * 100}%`, borderRadius: 8, background: i === sel ? 'var(--blue)' : 'var(--blue-tint)', minHeight: 8, transition: 'background 160ms' }} />)}
      </div>
      <div style={{ display: 'grid', gridTemplateColumns: 'repeat(7, 1fr)', gap: 8, paddingTop: 8 }}>{days.map((d, i) => <span key={d.d} className="t-meta" style={{ textAlign: 'center', color: i === sel ? 'var(--blue-deep)' : undefined, fontWeight: i === sel ? 700 : 400 }}>{d.d}</span>)}</div>
    </div>
  );
}

function AnalyticsScreen() {
  const nav = useNav();
  const loaded = useLoaded(450);
  const a = window.V2.analytics;
  const { fmt, money, creators } = window.V2;
  const [range, setRange] = React.useState('This week');
  const [tab, setTab] = React.useState('Creators');
  const up = Math.round(((a.week.views - a.week.prevViews) / a.week.prevViews) * 100);
  const topCreators = creators.slice().sort((x, y) => y.views - x.views);
  return (
    <Screen title="Analytics" sub={loaded ? <Chip tone="green" icon="trending_up">Up {up}% this week</Chip> : <span style={{ height: 26 }} />} right={<IconButton name="settings" label="Settings" onClick={() => nav.push('Settings')} />}>
      <Chips options={['This week', 'Last week', 'This month', 'All time']} value={range} onChange={setRange} />
      <div style={{ height: 12 }} />
      {!loaded ? <><Skeleton h={150} /><Skeleton h={220} /><Skeleton h={200} /></> : <>
        <Card style={{ background: 'linear-gradient(140deg, #1BA6EE, #0A7FC2)', boxShadow: '0 8px 24px rgba(27,166,238,0.35)', color: '#fff' }}>
          <div style={{ font: '600 15px/20px var(--font)', opacity: 0.9 }}>Your creators made</div>
          <div style={{ font: '700 40px/46px var(--font)', letterSpacing: '-1px', marginTop: 2 }}>{money(a.week.sales)}</div>
          <div style={{ font: '400 13px/18px var(--font)', opacity: 0.85, marginTop: 4 }}>{money(a.week.sales - a.week.prevSales)} more than last week · Stripe since {a.stripeSince}</div>
          <div style={{ display: 'grid', gridTemplateColumns: 'repeat(3, 1fr)', gap: 12, paddingTop: 20 }}>
            {[[fmt(a.week.views), 'views'], [a.week.posts, 'posts live'], [a.week.signups, 'sign ups']].map(([v, l]) => <div key={l}><div style={{ font: '700 22px/26px var(--font)' }}>{v}</div><div style={{ font: '400 13px/18px var(--font)', opacity: 0.85 }}>{l}</div></div>)}
          </div>
        </Card>
        <Card><div className="t-card" style={{ marginBottom: 12 }}>Views by day</div><Bars days={a.days} unit="views" /></Card>
        <Seg options={[{ label: 'Creators' }, { label: 'Posts' }, { label: 'Formats' }]} value={tab} onChange={setTab} style={{ marginBottom: 12 }} />
        {tab === 'Creators' ? <Card pad={'4px 16px'}>{topCreators.map((c, i) => (
          <Row key={c.id} last={i === topCreators.length - 1} chevron onClick={() => nav.push('Creator', { id: c.id })}>
            <span className="t-meta" style={{ width: 14, fontWeight: 700, color: i === 0 ? 'var(--blue)' : undefined }}>{i + 1}</span><Avatar name={c.name} photo={c.photo} tone={c.tone} size={40} />
            <span style={{ flex: 1, minWidth: 0 }}><span className="t-row" style={{ display: 'block' }}>{c.name}</span><span className="t-meta">{c.posts} posts</span></span>
            <Num small align="right" value={fmt(c.views)} label="views" /><Num small align="right" value={money(c.earned)} label="earned" tone="green" style={{ width: 64 }} />
          </Row>
        ))}</Card> : null}
        {tab === 'Posts' ? <Card pad={'4px 16px'}>{a.topPosts.map((p, i) => (
          <Row key={p.id} last={i === a.topPosts.length - 1} chevron onClick={() => nav.push('Brief', { id: p.id })}>
            <span className="t-meta" style={{ width: 14, fontWeight: 700, color: i === 0 ? 'var(--blue)' : undefined }}>{i + 1}</span><Thumb w={40} post={p} title={false} radius={8} />
            <span style={{ flex: 1, minWidth: 0 }}><span className="t-row" style={{ display: 'block', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{p.title}</span><span className="t-meta">{p.creatorName} · {p.format}</span></span>
            <Num small align="right" value={fmt(p.views)} label="views" /><Num small align="right" value={money(p.sales)} label="sales" tone="green" style={{ width: 48 }} />
          </Row>
        ))}</Card> : null}
        {tab === 'Formats' ? <Card pad={'4px 16px'}>{a.formats.map((f, i) => (
          <Row key={f.name} last={i === a.formats.length - 1}>
            <span style={{ width: 40, height: 40, borderRadius: 12, background: TONES[FORMAT_TONE[f.name]][1], display: 'flex', alignItems: 'center', justifyContent: 'center' }}><Icon name={f.name === 'Reel' ? 'movie' : 'photo_library'} size={22} color={TONES[FORMAT_TONE[f.name]][2]} /></span>
            <span style={{ flex: 1 }}><span className="t-row" style={{ display: 'block' }}>{f.name}</span><span className="t-meta">{f.posts} posts</span></span>
            <Num small align="right" value={fmt(f.views)} label="views" /><Num small align="right" value={money(f.sales)} label="sales" tone="green" style={{ width: 60 }} /><Num small align="right" value={fmt(Math.round(f.views / f.posts))} label="a post" style={{ width: 52 }} />
          </Row>
        ))}</Card> : null}
      </>}
    </Screen>
  );
}
