/* One stack per tab. Push slides from the right, modal from the bottom, pop reverses. */
function Shell({ tabs, screens, initialTab = 'review' }) {
  const [tab, setTabRaw] = React.useState(initialTab);
  const [stacks, setStacks] = React.useState(() => Object.fromEntries(tabs.filter((t) => t.root).map((t) => [t.key, [{ id: 0, name: t.root, params: {} }]])));
  const seq = React.useRef(1);
  const stack = stacks[tab];
  const push = (name, params = {}, opts = {}) => { haptic(); setStacks((s) => ({ ...s, [tab]: [...s[tab], { id: seq.current++, name, params, modal: !!opts.modal }] })); };
  const pop = () => { haptic(); setStacks((s) => { const st = s[tab]; if (st.length < 2) return s; const last = st[st.length - 1]; if (last.out) return s; return { ...s, [tab]: [...st.slice(0, -1), { ...last, out: true }] }; }); };
  const popAll = () => setStacks((s) => { const st = s[tab]; if (st.length < 2) return s; return { ...s, [tab]: [st[0], { ...st[st.length - 1], out: true }] }; });
  const remove = (id) => setStacks((s) => ({ ...s, [tab]: s[tab].filter((e) => e.id !== id) }));
  const setTab = (k) => { haptic(); const t = tabs.find((x) => x.key === k); if (t.action) { t.action({ push }); return; } if (k === tab) popAll(); else setTabRaw(k); };
  const top = stack.filter((e) => !e.out).length - 1;
  const barVisible = stack.filter((e) => !e.out).length === 1;
  const nav = { push, pop, popAll, tab, setTab, barVisible };
  return (
    <NavCtx.Provider value={nav}>
      <div style={{ position: 'absolute', inset: 0, overflow: 'hidden', background: 'var(--bg)' }}>
        {stack.map((e, i) => {
          const S = screens[e.name];
          const covered = i < top && !stack[i + 1]?.modal;
          const anim = i === 0 ? 'none' : e.out ? (e.modal ? 'v2-sheet-out' : 'v2-pop') + ' var(--t-push) var(--ease) forwards' : (e.modal ? 'v2-sheet' : 'v2-push') + ' var(--t-push) var(--ease)';
          return (
            <div key={e.id} onAnimationEnd={() => e.out && remove(e.id)} style={{ position: 'absolute', inset: 0, background: 'var(--bg)', display: 'flex', flexDirection: 'column', animation: anim, transform: covered ? 'translateX(-24%)' : 'none', transition: 'transform var(--t-push) var(--ease)', zIndex: 10 + i, willChange: 'transform', overflow: 'hidden', boxShadow: i > 0 ? '-8px 0 24px rgba(16,24,40,0.12)' : 'none' }}>
              <S {...e.params} />
            </div>
          );
        })}
        <TabBar tabs={tabs} tab={tab} onTab={setTab} visible={barVisible} />
      </div>
    </NavCtx.Provider>
  );
}

const BAR_H = 84;
function TabBar({ tabs, tab, onTab, visible }) {
  // Notch: 72 wide, 32 deep, 12px eased shoulders each side. One continuous path, top corners r24.
  const W = 390, H = BAR_H, cx = W / 2, nw = 72, nd = 32, sh = 12, cr = 24;
  const l = cx - nw / 2, r = cx + nw / 2;
  const notch = [
    `M0 ${cr}`, `A${cr} ${cr} 0 0 1 ${cr} 0`, `H${l - sh}`,
    `C${l - sh / 3} 0 ${l - sh / 3} 4 ${l} 6`,
    `C${l + 4} ${nd * 0.7} ${l + nw * 0.25} ${nd} ${cx} ${nd}`,
    `C${r - nw * 0.25} ${nd} ${r - 4} ${nd * 0.7} ${r} 6`,
    `C${r + sh / 3} 4 ${r + sh / 3} 0 ${r + sh} 0`,
    `H${W - cr}`, `A${cr} ${cr} 0 0 1 ${W} ${cr}`, `V${H + 60}H0Z`,
  ].join(' ');
  const [pressed, setPressed] = React.useState(false);
  return (
    <div style={{ position: 'absolute', left: 0, right: 0, bottom: 0, height: H, display: visible ? 'block' : 'none', zIndex: 100 }}>
      <svg width="100%" height={H} viewBox={`0 0 ${W} ${H}`} preserveAspectRatio="none" style={{ position: 'absolute', inset: 0, display: 'block', overflow: 'visible', filter: 'drop-shadow(0 -2px 12px rgba(0,0,0,0.06))' }}><path d={notch} fill="#fff" /></svg>
      <div style={{ position: 'absolute', inset: 0, display: 'grid', gridTemplateColumns: 'repeat(5, 1fr)' }}>
        {tabs.map((t) => {
          const on = t.key === tab;
          const color = on ? '#0A4DA3' : '#1BA6EE';
          return (
            <button key={t.key} type="button" className="p" onClick={() => onTab(t.key)} onPointerDown={() => t.action && setPressed(true)} onPointerUp={() => setPressed(false)} onPointerLeave={() => setPressed(false)} style={{ position: 'relative', display: 'flex', flexDirection: 'column', alignItems: 'center', justifyContent: 'flex-start', paddingTop: 10, gap: 4, color }}>
              {on && !t.action ? <span style={{ position: 'absolute', left: '50%', top: 4, width: 64, height: 60, marginLeft: -32, borderRadius: 999, background: 'var(--blue-tint)', zIndex: -1 }} /> : null}
              {t.action ? <span style={{ position: 'absolute', top: -40, left: '50%', width: 64, height: 64, marginLeft: -32, borderRadius: 999, background: '#1BA6EE', border: '5px solid #fff', boxSizing: 'border-box', boxShadow: '0 4px 12px rgba(16,24,40,0.14)', display: 'flex', alignItems: 'center', justifyContent: 'center', transform: pressed ? 'scale(0.94)' : 'scale(1)', transition: 'transform 120ms ease' }}><Icon name="add" size={30} color="#fff" weight={700} /></span> : <span style={{ position: 'relative' }}><Icon name={t.icon} size={26} weight={500} fill={1} color={color} />{t.count ? <span style={{ position: 'absolute', top: -6, right: -10, minWidth: 18, height: 18, padding: '0 5px', borderRadius: 999, background: '#1BA6EE', border: '2px solid #fff', boxSizing: 'content-box', color: '#fff', font: '700 11px/18px var(--font)', textAlign: 'center' }}>{t.count}</span> : null}</span>}
              <span style={{ font: '600 12px/14px var(--font)', color: t.action ? '#1BA6EE' : color, marginTop: t.action ? 30 : 0 }}>{t.label}</span>
            </button>
          );
        })}
      </div>
    </div>
  );
}

/* Screen: 32 bold title at a tab root, back chevron on a push, content scrolls under a pinned footer. */
function Screen({ title, back, right, children, footer, sub, scrollRef, white }) {
  const nav = useNav();
  const barOn = nav && nav.barVisible && !back;
  const padBottom = barOn ? 24 : footer ? 24 : 40;
  return (
    <div style={{ position: 'absolute', inset: 0, bottom: barOn ? BAR_H + 36 : 0, display: 'flex', flexDirection: 'column', paddingTop: 54, background: white ? 'var(--card)' : 'var(--bg)' }}>
      {back ? (
        <div style={{ display: 'flex', alignItems: 'center', height: 52, padding: '0 8px 0 4px', flexShrink: 0 }}>
          <button type="button" className="p" aria-label="Back" onClick={nav.pop} style={{ width: 44, height: 44, display: 'flex', alignItems: 'center', justifyContent: 'center' }}><Icon name="chevron_left" size={32} color="var(--ink)" /></button>
          <div className="t-row" style={{ flex: 1, minWidth: 0, whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis', marginLeft: -2 }}>{title}</div>
          {right ? <div style={{ display: 'flex', alignItems: 'center', gap: 4, paddingRight: 8 }}>{right}</div> : null}
        </div>
      ) : null}
      <div ref={scrollRef} className="v2-scroll" style={{ flex: 1, minHeight: 0, overflowY: 'auto', padding: `0 var(--screen) ${padBottom}px` }}>

  );

function Empty({ children }) { return <Card style={{ textAlign: 'center' }}><span className="t-body" style={{ color: 'var(--grey-4)' }}>{children}</span></Card>; }

function Skeleton({ h = 56, w = '100%', r = 16, style }) { return <div style={{ height: h, width: w, borderRadius: r, background: 'var(--card)', boxShadow: 'var(--shadow-card)', animation: 'v2-shimmer 1.2s ease-in-out infinite', marginBottom: 12, ...style }} />; }

/* Bottom sheet. */
function Sheet({ open, onClose, title, children, footer }) {
  const [shown, setShown] = React.useState(open);
  const [closing, setClosing] = React.useState(false);
  React.useEffect(() => { if (open) { setShown(true); setClosing(false); } else if (shown) { setClosing(true); } }, [open]);
  if (!shown) return null;
  return (
    <div style={{ position: 'absolute', inset: 0, zIndex: 200 }}>
      <div onClick={onClose} style={{ position: 'absolute', inset: 0, background: 'var(--scrim)', animation: closing ? 'v2-scrim var(--t-push) reverse forwards' : 'v2-scrim var(--t-push)' }} />
      <div onAnimationEnd={() => { if (closing) { setShown(false); setClosing(false); } }} style={{ position: 'absolute', left: 0, right: 0, bottom: 0, background: 'var(--card)', borderRadius: '20px 20px 0 0', boxShadow: 'var(--shadow-float)', padding: '12px 20px 34px', animation: `${closing ? 'v2-sheet-out' : 'v2-sheet'} var(--t-push) var(--ease) forwards`, maxHeight: '86%', display: 'flex', flexDirection: 'column' }}>
        <div style={{ width: 36, height: 4, borderRadius: 999, background: 'var(--grey-2)', margin: '0 auto 16px' }} />
        {title ? <div className="t-card" style={{ fontSize: 24, lineHeight: '30px', marginBottom: 16 }}>{title}</div> : null}
        <div className="v2-scroll" style={{ overflowY: 'auto', flex: 1, minHeight: 0 }}>{children}</div>
        {footer ? <div style={{ display: 'flex', gap: 8, paddingTop: 16 }}>{footer}</div> : null}
      </div>
    </div>
  );
}

/* Selectable option: tint fill when on. */
function Option({ on, onClick, children, meta }) {
  return (
    <button type="button" className="p" onClick={() => { haptic(); onClick(); }} style={{ display: 'flex', alignItems: 'center', gap: 12, width: '100%', textAlign: 'left', padding: '0 16px', minHeight: 56, borderRadius: 14, background: on ? 'var(--blue-tint)' : 'var(--grey-1)', color: on ? 'var(--blue-deep)' : 'var(--ink)', marginBottom: 8, boxShadow: on ? 'inset 0 0 0 2px var(--blue)' : 'none' }}>
      <span style={{ flex: 1 }}><span className="t-row" style={{ color: 'inherit', display: 'block' }}>{children}</span>{meta ? <span className="t-meta" style={{ display: 'block' }}>{meta}</span> : null}</span>
      <span style={{ width: 24, height: 24, borderRadius: 999, background: on ? 'var(--blue)' : 'var(--grey-2)', display: 'flex', alignItems: 'center', justifyContent: 'center' }}>{on ? <Icon name="check" size={16} color="#fff" weight={700} /> : null}</span>
    </button>
  );
}

function Field({ label, value, onChange, placeholder, multiline, right, style }) {
  const base = { width: '100%', border: 'none', outline: 'none', background: 'transparent', font: '400 15px/20px var(--font)', color: 'var(--ink)', padding: 0, resize: 'none' };
  return (
    <label style={{ display: 'block', marginBottom: 12, ...style }}>
      {label ? <span className="t-meta" style={{ display: 'block', marginBottom: 6 }}>{label}</span> : null}
      <span className="card" style={{ display: 'flex', alignItems: 'center', gap: 8, padding: '14px 16px' }}>
        {multiline ? <textarea rows={3} value={value} placeholder={placeholder} onChange={(e) => onChange(e.target.value)} style={base} /> : <input value={value} placeholder={placeholder} onChange={(e) => onChange(e.target.value)} style={base} />}
        {right}
      </span>
    </label>
  );
}
