/* Briefs tab: week cards or calendar, week detail with all 21 posts, brief detail, new brief, ideas. */
const briefStore = { weeks: window.V2.weeks, listeners: new Set() };
const useBriefs = () => { const [, tick] = React.useState({}); React.useEffect(() => { briefStore.listeners.add(tick); return () => briefStore.listeners.delete(tick); }, []); return briefStore; };
const bumpBriefs = () => briefStore.listeners.forEach((l) => l({}));
const weekChip = (w) => w.status === 'current' ? <Chip tone="green" icon="bolt">Day {w.day} of 7</Chip> : w.status === 'next' ? <Chip tone="amber" icon="edit_calendar">Plan by Sat</Chip> : <Chip tone="grey">Done</Chip>;

function WeekCard({ w, onClick }) {
  const reels = w.posts.filter((p) => p.format === 'Reel'), slides = w.posts.filter((p) => p.format === 'Slideshow');
  const done = (l) => l.filter((p) => p.state === 'Posted').length;
  const bar = (icon, n, d) => <span style={{ display: 'flex', alignItems: 'center', gap: 10, flex: 1, minWidth: 0 }}><Icon name={icon} size={20} fill={0} weight={300} color="var(--grey-4)" /><span style={{ flex: 1, height: 5, borderRadius: 999, background: 'var(--grey-1)', overflow: 'hidden' }}><span style={{ display: 'block', width: `${d ? (n / d) * 100 : 0}%`, height: '100%', background: 'var(--blue)', borderRadius: 999 }} /></span><span style={{ font: '600 15px/20px var(--font)', color: 'var(--grey-5, #3C4452)', whiteSpace: 'nowrap' }}>{n}/{d}</span></span>;
  const planned = w.status !== 'next';
  return (
    <Card onClick={onClick} pad={'18px 20px 20px'}>
      <div style={{ display: 'flex', alignItems: 'center', gap: 10 }}>
        <span style={{ font: '700 21px/26px var(--font)', color: 'var(--ink)', whiteSpace: 'nowrap', letterSpacing: -0.3 }}>Week {w.n}</span>
        <span style={{ font: '400 15px/20px var(--font)', color: 'var(--grey-4)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis', flex: 1, minWidth: 0 }}>{w.range}</span>
        {w.status === 'current' ? <Chip tone="green">Day {w.day} of 7</Chip> : w.status === 'next' ? <Chip tone="blue">Next week</Chip> : <Chip tone="grey">Done</Chip>}
        <Icon name="chevron_right" size={20} weight={400} color="var(--grey-3)" />
      </div>
      {planned ? <div style={{ display: 'flex', gap: 18, paddingTop: 16 }}>{bar('videocam', done(reels), reels.length)}{bar('imagesmode', done(slides), slides.length)}</div>
        : <div style={{ font: '400 16px/22px var(--font)', color: 'var(--grey-3)', paddingTop: 12 }}>Not planned yet. Tap to start it.</div>}
    </Card>
  );
}

function PostRow({ p, i, last, onClick }) {
  const { fmt, money } = window.V2;
  const open = p.state === 'Open';
  return (
    <Row last={last} top={10} bottom={10} onClick={onClick}>
      <Thumb w={40} post={p} title={false} radius={8} style={open ? { background: 'var(--grey-1)', opacity: 1 } : undefined} />
      <span style={{ flex: 1, minWidth: 0 }}>
        <span className="t-row" style={{ display: 'block', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis', color: open ? 'var(--blue)' : undefined }}>{open ? 'Write this brief' : p.title}</span>
        <span style={{ display: 'flex', gap: 6, alignItems: 'center', marginTop: 4 }}><Chip tone={STATE_TONE[p.state]} style={{ height: 22, fontSize: 12 }}>{p.state}</Chip><span className="t-meta" style={{ whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>Day {p.day} · {p.creatorName} · {p.kind}</span></span>
      </span>
      {p.state === 'Posted' ? <><Num small align="right" value={fmt(p.views)} label="views" /><Num small align="right" value={money(p.sales)} label="sales" tone="green" style={{ width: 44 }} /></> : <Icon name="chevron_right" size={22} color="var(--grey-3)" />}
    </Row>
  );
}

const TYPES = { Reel: ['Talking head', 'Numbered list', 'How to', 'Explainer', 'Storytime', 'Green screen react', 'Long text'], Slideshow: ['How to', 'Numbered tips', 'Getting started', 'Myths'] };

const STEPS = [
  { key: 'format', title: 'What are we making?', options: [['Post', 'videocam', 'A video the creator records'], ['Slideshow', 'photo_library', 'Images with text, swiped through']] },
  { key: 'kind', title: 'How much do you script?', options: [['Guided', 'format_list_numbered', 'You write every point'], ['Outlined', 'segment', 'You set the hook and close'], ['Unscripted', 'auto_awesome', 'The creator makes what they want']] },
  { key: 'type', title: 'Post type', options: [['Talking Head', 'record_voice_over', 'Creator to camera'], ['Green Screen', 'preview', 'Creator over a screenshot or clip'], ['Long Text', 'notes', 'Text on screen, voiceover under it']] },
];
function NewBriefScreen({ n = 5, slot }) {
  const nav = useNav();
  const key = 'noni.brief.' + slot;
  const load = () => { try { return JSON.parse(localStorage.getItem(key) || 'null'); } catch (e) { return null; } };
  const [st, setSt] = React.useState(() => load() || { step: 0 });
  React.useEffect(() => { localStorage.setItem(key, JSON.stringify(st)); }, [st]);
  const step = STEPS[st.step];
  const pick = (v) => { haptic(); setSt((x) => ({ ...x, [step.key]: v })); };
  const next = () => setSt((x) => ({ ...x, step: Math.min(STEPS.length - 1, x.step + 1) }));
  const back = () => st.step === 0 ? nav.pop() : setSt((x) => ({ ...x, step: x.step - 1 }));
  const val = st[step.key];
  const done = st.step === STEPS.length - 1;
  return (
    <div style={{ position: 'absolute', inset: 0, display: 'flex', flexDirection: 'column', paddingTop: 54, background: 'var(--bg)' }}>
      <div style={{ display: 'flex', alignItems: 'center', height: 52, padding: '0 8px 0 4px', flexShrink: 0 }}>
        <button type="button" className="p" aria-label="Back" onClick={back} style={{ width: 44, height: 44, display: 'flex', alignItems: 'center', justifyContent: 'center' }}><Icon name={st.step === 0 ? 'close' : 'chevron_left'} size={st.step === 0 ? 26 : 32} color="var(--ink)" /></button>
        <span style={{ flex: 1, display: 'flex', justifyContent: 'center', gap: 6 }}>{STEPS.map((_, i) => <span key={i} style={{ width: i === st.step ? 24 : 8, height: 8, borderRadius: 999, background: i <= st.step ? 'var(--blue)' : 'var(--grey-2)', transition: 'width 200ms' }} />)}</span>
        <span style={{ width: 44 }} />
      </div>
      <div className="v2-scroll" style={{ flex: 1, minHeight: 0, overflowY: 'auto', padding: '8px 20px 24px' }}>
        <div className="t-meta" style={{ fontWeight: 600, color: 'var(--grey-4)' }}>Step {st.step + 1} of {STEPS.length}</div>
        <h1 className="t-h" style={{ margin: '6px 0 20px' }}>{step.title}</h1>
        {step.options.map(([label, ic, sub]) => { const on = val === label; return (
          <Card key={label} onClick={() => pick(label)} style={{ display: 'flex', alignItems: 'center', gap: 14, boxShadow: on ? '0 0 0 2px var(--blue), var(--shadow-card)' : 'var(--shadow-card)' }}>
            <span style={{ width: 48, height: 48, borderRadius: 14, background: on ? 'var(--blue)' : 'var(--blue-tint)', display: 'flex', alignItems: 'center', justifyContent: 'center', flex: '0 0 auto' }}><Icon name={ic} size={24} color={on ? '#fff' : 'var(--blue-deep)'} /></span>
            <span style={{ flex: 1, minWidth: 0 }}><span className="t-row" style={{ display: 'block' }}>{label}</span><span className="t-meta" style={{ display: 'block', marginTop: 2 }}>{sub}</span></span>
            <span style={{ width: 24, height: 24, borderRadius: 999, border: on ? 'none' : '2px solid var(--grey-2)', background: on ? 'var(--blue)' : 'transparent', display: 'flex', alignItems: 'center', justifyContent: 'center', flex: '0 0 auto' }}>{on ? <Icon name="check" size={16} weight={700} color="#fff" /> : null}</span>
          </Card>); })}
      </div>
      <div style={{ flexShrink: 0, padding: '12px 20px 34px', display: 'flex', background: 'var(--card)', boxShadow: '0 -1px 0 var(--grey-2)' }}><Button disabled={!val} onClick={done ? nav.pop : next}>{done ? 'Continue' : 'Next'}</Button></div>
    </div>
  );
}
